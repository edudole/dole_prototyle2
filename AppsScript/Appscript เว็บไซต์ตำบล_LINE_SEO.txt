function normalizeTemplateSettingName_(value) {
  const match = String(value || '').trim().match(/(\d+)/);
  const n = match ? Number(match[1]) : 1;
  return n >= 1 && n <= 9 ? 'Template' + n : TEMPLATE_SETTING_DEFAULT;
}

function getTemplateSettingSiteKey_() {
  try {
    const sh = SpreadsheetApp.openById(ID_SHEET).getSheetByName('Sheet1');
    const key = sh ? String(sh.getRange('C9').getDisplayValue() || '').trim() : '';
    return 'tambon:' + (key || ID_SHEET);
  } catch (ignore) {
    return 'tambon:' + ID_SHEET;
  }
}

function getTemplateSetting_() {
  const siteKey = getTemplateSettingSiteKey_();
  const cache = CacheService.getScriptCache();
  const cacheKey = 'LP360_TEMPLATE_' + Utilities.base64EncodeWebSafe(
    Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, siteKey)
  ).replace(/=+$/,'');
  const cached = String(cache.get(cacheKey) || '').trim();
  if (cached) return { template: normalizeTemplateSettingName_(cached), siteKey: siteKey };

  const ss = getTemplateSettingSpreadsheet_();
  const sh = ss.getSheetByName(TEMPLATE_SETTING_SHEET_NAME);
  if (!sh || sh.getLastRow() < 2) {
    return { template: TEMPLATE_SETTING_DEFAULT, siteKey: siteKey };
  }

  const rows = sh.getRange(2, 1, sh.getLastRow() - 1, 2).getDisplayValues();
  for (let i = 0; i < rows.length; i++) {
    if (String(rows[i][0] || '').trim() === siteKey) {
      const template = normalizeTemplateSettingName_(rows[i][1]);
      try { cache.put(cacheKey, template, 300); } catch (ignore) {}
      return { template: template, siteKey: siteKey };
    }
  }
  return { template: TEMPLATE_SETTING_DEFAULT, siteKey: siteKey };
}

function saveTemplateSetting_(value) {
  const template = normalizeTemplateSettingName_(value);
  const siteKey = getTemplateSettingSiteKey_();
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const ss = getTemplateSettingSpreadsheet_();
    let sh = ss.getSheetByName(TEMPLATE_SETTING_SHEET_NAME);
    if (!sh) {
      sh = ss.insertSheet(TEMPLATE_SETTING_SHEET_NAME);
      sh.getRange(1, 1, 1, 3).setValues([['site_key', 'template', 'updated_at']]);
      sh.setFrozenRows(1);
    }

    const lastRow = sh.getLastRow();
    let targetRow = 0;
    if (lastRow >= 2) {
      const keys = sh.getRange(2, 1, lastRow - 1, 1).getDisplayValues();
      for (let i = 0; i < keys.length; i++) {
        if (String(keys[i][0] || '').trim() === siteKey) {
          targetRow = i + 2;
          break;
        }
      }
    }
    if (!targetRow) targetRow = Math.max(2, lastRow + 1);
    sh.getRange(targetRow, 1, 1, 3).setValues([[siteKey, template, new Date()]]);
    const cacheKey = 'LP360_TEMPLATE_' + Utilities.base64EncodeWebSafe(
      Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, siteKey)
    ).replace(/=+$/,'');
    try { CacheService.getScriptCache().put(cacheKey, template, 300); } catch (ignore) {}
    return { template: template, siteKey: siteKey, updatedAt: new Date().toISOString() };
  } finally {
    lock.releaseLock();
  }
}



// Color theme selection is stored per site and per Template.
const TEMPLATE_COLOR_SETTING_SHEET_NAME = 'template_color_setting';
function getTemplateColorSetting_(templateValue) {
  const template = normalizeTemplateSettingName_(templateValue);
  const siteKey = getTemplateSettingSiteKey_();
  const sheet = getTemplateSettingSpreadsheet_().getSheetByName(TEMPLATE_COLOR_SETTING_SHEET_NAME);
  if (!sheet || sheet.getLastRow() < 2) return { template: template, palette: 1 };
  const rows = sheet.getRange(2, 1, sheet.getLastRow() - 1, 3).getDisplayValues();
  for (let i = 0; i < rows.length; i++) {
    if (String(rows[i][0]) === siteKey && String(rows[i][1]) === template) {
      const palette = Number(rows[i][2]);
      return { template: template, palette: palette >= 1 && palette <= 6 ? palette : 1 };
    }
  }
  return { template: template, palette: 1 };
}
function saveTemplateColorSetting_(templateValue, paletteValue) {
  const template = normalizeTemplateSettingName_(templateValue);
  const palette = Number(paletteValue);
  if (!Number.isInteger(palette) || palette < 1 || palette > 6) throw new Error('ธีมสีต้องเป็น 1–6');
  const siteKey = getTemplateSettingSiteKey_();
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const ss = getTemplateSettingSpreadsheet_();
    let sheet = ss.getSheetByName(TEMPLATE_COLOR_SETTING_SHEET_NAME);
    if (!sheet) {
      sheet = ss.insertSheet(TEMPLATE_COLOR_SETTING_SHEET_NAME);
      sheet.getRange(1, 1, 1, 4).setValues([['site_key', 'template', 'palette', 'updated_at']]);
      sheet.setFrozenRows(1);
    }
    const last = sheet.getLastRow();
    let row = last + 1;
    if (last >= 2) {
      const keys = sheet.getRange(2, 1, last - 1, 2).getDisplayValues();
      for (let i = 0; i < keys.length; i++) {
        if (String(keys[i][0]) === siteKey && String(keys[i][1]) === template) { row = i + 2; break; }
      }
    }
    sheet.getRange(row, 1, 1, 4).setValues([[siteKey, template, palette, new Date()]]);
    return { template: template, palette: palette };
  } finally { lock.releaseLock(); }
}

const HERO_CONTENT_SETTING_SHEET_NAME = 'hero_content_setting';

function heroContentNumber_(value, fallback, min, max) {
  const n = Number(value);
  if (!Number.isFinite(n)) return fallback;
  return Math.max(min, Math.min(max, n));
}

function heroContentColor_(value, fallback) {
  const v = String(value || '').trim();
  return /^#[0-9a-fA-F]{6}$/.test(v) ? v.toLowerCase() : fallback;
}

function heroContentBool_(value) {
  if (value === false || value === 0) return false;
  const v = String(value == null ? '' : value).trim().toLowerCase();
  return !(v === 'false' || v === '0' || v === 'no' || v === 'off');
}

function heroContentCacheKey_(siteKey, template) {
  const raw = String(siteKey || '') + '|' + String(template || '');
  return 'LP360_HERO_' + Utilities.base64EncodeWebSafe(
    Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, raw)
  ).replace(/=+$/, '');
}

function getHeroContentSetting_(templateValue) {
  const siteKey = getTemplateSettingSiteKey_();
  const template = normalizeTemplateSettingName_(templateValue);
  const cache = CacheService.getScriptCache();
  const cacheKey = heroContentCacheKey_(siteKey, template);
  const cached = String(cache.get(cacheKey) || '').trim();
  if (cached) {
    try { return JSON.parse(cached); } catch (ignore) {}
  }

  const ss = getTemplateSettingSpreadsheet_();
  const sh = ss.getSheetByName(HERO_CONTENT_SETTING_SHEET_NAME);
  if (!sh || sh.getLastRow() < 2) {
    return { configured: false, siteKey: siteKey, template: template };
  }

  const rows = sh.getRange(2, 1, sh.getLastRow() - 1, 19).getDisplayValues();
  for (let i = 0; i < rows.length; i++) {
    if (String(rows[i][0] || '').trim() !== siteKey) continue;
    if (normalizeTemplateSettingName_(rows[i][1]) !== template) continue;
    const out = {
      configured: true,
      siteKey: siteKey,
      template: template,
      xPct: heroContentNumber_(rows[i][2], 5, 0, 98),
      yPct: heroContentNumber_(rows[i][3], 20, 0, 98),
      widthPct: heroContentNumber_(rows[i][4], 55, 20, 100),
      titleSize: heroContentNumber_(rows[i][5], 56, 12, 140),
      kickerSize: heroContentNumber_(rows[i][6], 12, 8, 56),
      descriptionSize: heroContentNumber_(rows[i][7], 16, 9, 72),
      align: ['left', 'center', 'right'].indexOf(String(rows[i][8] || '').trim()) >= 0 ? String(rows[i][8]).trim() : 'left',
      titleColor: heroContentColor_(rows[i][9], '#ffffff'),
      kickerColor: heroContentColor_(rows[i][10], '#ffffff'),
      descriptionColor: heroContentColor_(rows[i][11], '#ffffff'),
      visible: heroContentBool_(rows[i][12]),
      updatedAt: String(rows[i][13] || ''),
      titleWeight: String(rows[i][14] || '').trim().toLowerCase() === 'bold' ? 'bold' : (String(rows[i][14] || '').trim().toLowerCase() === 'normal' ? 'normal' : 'template'),
      titleStyle: String(rows[i][15] || '').trim().toLowerCase() === 'italic' ? 'italic' : (String(rows[i][15] || '').trim().toLowerCase() === 'normal' ? 'normal' : 'template'),
      overlayConfigured: /^#[0-9a-fA-F]{6}$/.test(String(rows[i][16] || '').trim()) && String(rows[i][17] || '').trim() !== '',
      overlayColor: heroContentColor_(rows[i][16], '#051c2c'),
      overlayOpacity: String(rows[i][17] || '').trim() === '' ? 72 : heroContentNumber_(rows[i][17], 72, 0, 100),
      overlayMode: ['left', 'right', 'uniform'].indexOf(String(rows[i][18] || '').trim().toLowerCase()) >= 0 ? String(rows[i][18]).trim().toLowerCase() : 'uniform'
    };
    try { cache.put(cacheKey, JSON.stringify(out), 300); } catch (ignore) {}
    return out;
  }
  return { configured: false, siteKey: siteKey, template: template };
}

function jsonOutput_(payload, callback) {
  const json = JSON.stringify(payload);
  const cb = String(callback || '').trim();

  if (cb) {
    const safeCallback = cb.replace(
      /[^a-zA-Z0-9_$\.]/g,
      ''
    );

    if (!safeCallback) {
      return ContentService
        .createTextOutput(JSON.stringify({
          success: false,
          message: 'Callback ไม่ถูกต้อง'
        }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService
      .createTextOutput(
        safeCallback + '(' + json + ');'
      )
      .setMimeType(
        ContentService.MimeType.JAVASCRIPT
      );
  }

  return ContentService
    .createTextOutput(json)
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  e = e || { parameter: {} };

  const p = e.parameter || {};
  const callback = String(p.callback || '').trim();

  const mode = String(p.mode || '')
    .trim()
    .toLowerCase();

  const page = String(p.page || '')
    .trim()
    .toLowerCase();

  // Critical UI setting: โหลด Template + Hero Content ในคำขอเดียวก่อนแสดงหน้าเว็บ
  if (mode === 'uicritical') {
    try {
      const templateData = getTemplateSetting_();
      const template = normalizeTemplateSettingName_(templateData && templateData.template);
      return jsonOutput_({
        success: true,
        data: {
          template: template,
          templateSetting: templateData,
          hero: getHeroContentSetting_(template)
        }
      }, callback);
    } catch (err) {
      return jsonOutput_({
        success: false,
        data: {
          template: TEMPLATE_SETTING_DEFAULT,
          hero: { configured: false, template: TEMPLATE_SETTING_DEFAULT }
        },
        message: err.message
      }, callback);
    }
  }

  // Template ที่เว็บไซต์กำลังใช้งาน (อ่านได้แบบ public เพื่อให้ทุกหน้าใช้รูปแบบเดียวกัน)
  if (mode === 'templatesetting') {
    try {
      return jsonOutput_({
        success: true,
        data: getTemplateSetting_()
      }, callback);
    } catch (err) {
      return jsonOutput_({
        success: false,
        data: { template: TEMPLATE_SETTING_DEFAULT },
        message: err.message
      }, callback);
    }
  }


  if (mode === 'templatecolorsetting') {
    try { return jsonOutput_({ success: true, data: getTemplateColorSetting_(p.template) }, callback); }
    catch (err) { return jsonOutput_({ success: false, message: err.message }, callback); }
  }

  // การจัดตำแหน่ง/ขนาด/สีของ hero-content (public read)
  if (mode === 'herocontentsetting') {
    try {
      return jsonOutput_({
        success: true,
        data: getHeroContentSetting_(p.template || TEMPLATE_SETTING_DEFAULT)
      }, callback);
    } catch (err) {
      return jsonOutput_({
        success: false,
        data: { configured: false, template: normalizeTemplateSettingName_(p.template) },
        message: err.message
      }, callback);
    }
  }

  // หน้าหลังบ้านครู
  if (page === 'teacher') {
    setupLearningBaseSheets_();

    return HtmlService.createTemplateFromFile('teacher')
      .evaluate()
      .setTitle('Dashboard ครูประจำฐาน')
      .addMetaTag(
        'viewport',
        'width=device-width, initial-scale=1'
      )
      .setXFrameOptionsMode(
        HtmlService.XFrameOptionsMode.ALLOWALL
      );
  }

  // หน้า Dashboard รายงานสถิติใบงาน/แบบทดสอบจากไฟล์ "ผู้นำ"
  if (mode === 'report') {
    return doGetReport_(e);
  }


  // SEO แบบ public สำหรับ GitHub Actions / LINE preview
  if (mode === 'seo') {
    try {
      return jsonOutput_({ success: true, data: getPublicSeoData_() }, callback);
    } catch (err) {
      return jsonOutput_({ success: false, message: err.message }, callback);
    }
  }

  // โหลดข้อมูลหลักของ index.html ในคำขอเดียว
  if (mode === 'homefast') {
    try {
      return jsonOutput_({
        success: true,
        data: getHomeFastDataCached_(String(p.fresh || '') === '1')
      }, callback);
    } catch (err) {
      return jsonOutput_({
        success: false,
        message: err.message
      }, callback);
    }
  }

  // announcement กลางของทุกเว็บไซต์: DATABSE_MAIN_ID > Sheet!B24 / D24
  if (mode === 'announcement') {
    try {
      return jsonOutput_({
        success: true,
        data: getCentralAnnouncement_()
      }, callback);
    } catch (err) {
      return jsonOutput_({ success: false, message: err.message }, callback);
    }
  }



  // ลำดับ/การมองเห็น SECTION ของ index.html (เก็บใน ScriptProperties เพื่ออ่านเร็ว)
  if (mode === 'sectionlayout') {
    try {
      return jsonOutput_({
        success: true,
        items: getSectionLayout_()
      }, callback);
    } catch (err) {
      return jsonOutput_({ success: false, items: [], message: err.message }, callback);
    }
  }

  // รวมระบบแหล่งเรียนรู้เข้ากับ Web App หลัก โดยไม่ทับ doGet เดิม
  const learningSourceModes = [
    'learningbootstrap',
    'learningareaoptions',
    'learningareas',
    'learningsheetoptions',
    'learningbox',
    'learningdetail',
    'learningreview'
  ];

  // ระบบแหล่งเรียนรู้
  // - /exec หรือ ?page=home  -> หน้าบ้าน (Index)
  // - ?page=admin           -> หลังบ้าน (Dashboard)
  // - mode ของระบบแหล่งเรียนรู้ -> API/JSON เดิม
  if (page === 'admin') {
    return doGetLearningSource_(e);
  }

  if (!mode && (!page || page === 'home')) {
    return doGetLearningSource_(e);
  }

  if (learningSourceModes.indexOf(mode) !== -1) {
    return doGetLearningSource_(e);
  }

  // Top 3 สำหรับ student-service-card บน GitHub
  if (mode === 'studentservicetop3') {
    try {
      return jsonOutput_(getStudentServiceTop3_(), callback);
    } catch (err) {
      return jsonOutput_({
        success: false,
        worksheet: {},
        quiz: {},
        message: err.message
      }, callback);
    }
  }

  // กล่องสรุป homeSection บน GitHub (อยู่เหนือ newsSlides)
  if (mode === 'homesummary') {
    try {
      return jsonOutput_({
        success: true,
        data: getHomeSummaryData_()
      }, callback);
    } catch (err) {
      return jsonOutput_({
        success: false,
        message: err.message
      }, callback);

// Cliproom exec URL resolver
// DATABSE_MAIN_ID > Sheet!D17
// ============================================================
function getCliproomExecUrl_() {
  const cache = CacheService.getScriptCache();
  const cacheKey = 'CLIPROOM_EXEC_SUBDISTRICT_V3';
  const cached = cache.get(cacheKey);
  if (cached && /^https:\/\/script\.google\.com\/macros\/s\/[^/?#]+\/exec(?:[?#].*)?$/i.test(cached)) return cached;

  const ss = SpreadsheetApp.openById(DATABSE_MAIN_ID);
  const sheet = ss.getSheetByName('Sheet');
  if (!sheet) throw new Error('ไม่พบชีต Sheet');

  const url = String(sheet.getRange('D17').getDisplayValue() || '').trim();
  if (!url) throw new Error('ยังไม่ได้ระบุ URL Cliproom ที่ Sheet!D17');
  if (!/^https:\/\/script\.google\.com\/macros\/s\/[^/?#]+\/exec(?:[?#].*)?$/i.test(url)) {
    throw new Error('URL ที่ Sheet!D17 ต้องเป็น Google Apps Script /exec');
  }
  try { cache.put(cacheKey, url, 60); } catch (ignore) {}
  return url;
}

// ============================================================
// GitHub page -> Apps Script exec URL resolver
// quiz.html      = base!C15
// classroom.html = base!C14
// ============================================================
function getPageExecUrl_(target) {
  const key = String(target || '').trim().toLowerCase();
  const cellMap = {
    quiz: 'C15',
    classroom: 'C14'
  };
  const cell = cellMap[key];
  if (!cell) throw new Error('target ไม่ถูกต้อง: ใช้ quiz หรือ classroom');

  const ss = SpreadsheetApp.openById(DATABSE_MAIN_TEATHER_ID);
  const sheet = ss.getSheetByName('base');
  if (!sheet) throw new Error('ไม่พบชีต base');

  const url = String(sheet.getRange(cell).getDisplayValue() || '').trim();
  if (!url) throw new Error('ยังไม่ได้ระบุ URL ที่ base!' + cell);

  if (!/^https:\/\/script\.google\.com\/macros\/s\/[^/?#]+\/exec(?:[?#].*)?$/i.test(url)) {
    throw new Error('URL ที่ base!' + cell + ' ต้องเป็น Google Apps Script /exec');
  }
  return url;
}

// Z=สีข้อความ, AA=ลำดับ, AB=แสดง, AC1=รูปแบบปุ่ม, AC2=ค่ารูปแบบ
// ============================================================
const BUTTON_SECTION_CACHE_KEY = 'BUTTON_SECTION_V1';
const BUTTON_SECTION_STYLE_DEFAULT = 'icon-text';
const BUTTON_SECTION_STYLES = ['text','icon-text','icon-only','icon-top','compact'];

function getButtonSectionSheet_() {
  const ss = SpreadsheetApp.openById(DATABSE_MAIN_TEATHER_ID);
  const sheet = ss.getSheetByName('setting');
  if (!sheet) throw new Error('ไม่พบชีต setting');

  if (sheet.getMaxColumns() < 29) {
    sheet.insertColumnsAfter(sheet.getMaxColumns(), 29 - sheet.getMaxColumns());
  }

  const headers = ['ชื่อปุ่ม','รูปภาพ/ไอคอน','URL','สีปุ่ม','สีข้อความ','ลำดับ','แสดง','รูปแบบปุ่ม'];
  const current = sheet.getRange(1, 22, 1, 8).getDisplayValues()[0]; // V:AC
  const next = current.slice();
  headers.forEach(function(header, index) {
    if (!String(next[index] || '').trim()) next[index] = header;
  });
  sheet.getRange(1, 22, 1, 8).setValues([next]);
  return sheet;
}

function normalizeButtonSectionStyle_(value) {
  const style = String(value || '').trim().toLowerCase();
  return BUTTON_SECTION_STYLES.indexOf(style) !== -1 ? style : BUTTON_SECTION_STYLE_DEFAULT;
}

function normalizeButtonColor_(value, fallback) {
  const color = String(value || '').trim();
  return /^#[0-9a-f]{6}$/i.test(color) ? color : fallback;
}

function isButtonSectionHttpUrl_(value) {
  return /^https?:\/\//i.test(String(value || '').trim());
}

function getButtonSectionData_(forceRefresh) {
  const cache = CacheService.getScriptCache();
  if (!forceRefresh) {
    const cached = cache.get(BUTTON_SECTION_CACHE_KEY);
    if (cached) {
      try { return JSON.parse(cached); } catch (ignore) {}
    }
  }

  const sheet = getButtonSectionSheet_();
  const style = normalizeButtonSectionStyle_(sheet.getRange('AC2').getDisplayValue());
  const lastRow = sheet.getLastRow();
  const rows = lastRow >= 2
    ? sheet.getRange(2, 22, lastRow - 1, 7).getDisplayValues() // V:AB
    : [];

  const items = rows.map(function(row, index) {
    const visibleText = String(row[6] || '').trim().toLowerCase();
    return {
      name: String(row[0] || '').trim(),
      iconUrl: String(row[1] || '').trim(),
      url: String(row[2] || '').trim(),
      buttonColor: normalizeButtonColor_(row[3], '#2563eb'),
      textColor: normalizeButtonColor_(row[4], '#ffffff'),
      order: Number(row[5]) || (index + 1),
      visible: visibleText !== 'false' && visibleText !== '0' && visibleText !== 'ไม่'
    };
  }).filter(function(item) {
    return item.name || item.iconUrl || item.url;
  }).sort(function(a, b) {
    return a.order - b.order;
  });

  const data = { style: style, items: items };
  try { cache.put(BUTTON_SECTION_CACHE_KEY, JSON.stringify(data), 120); } catch (ignore) {}
  return data;
}

function saveButtonSectionData_(data) {
  data = data || {};
  const style = normalizeButtonSectionStyle_(data.style);
  const input = Array.isArray(data.items) ? data.items : [];
  if (input.length > 80) throw new Error('เพิ่มปุ่มได้สูงสุด 80 ปุ่ม');
  const iconRequired = style !== 'text';

  const items = input.map(function(raw, index) {
    raw = raw || {};
    const name = String(raw.name || '').trim().slice(0, 80);
    const iconUrl = String(raw.iconUrl || '').trim();
    const url = String(raw.url || '').trim();
    const buttonColor = normalizeButtonColor_(raw.buttonColor, '#2563eb');
    const textColor = normalizeButtonColor_(raw.textColor, '#ffffff');
    const visible = raw.visible !== false && String(raw.visible).toLowerCase() !== 'false';

    if (!name) throw new Error('ปุ่มลำดับที่ ' + (index + 1) + ' ยังไม่ได้ระบุชื่อปุ่ม');
    if (!isButtonSectionHttpUrl_(url)) throw new Error('URL ของปุ่ม "' + name + '" ไม่ถูกต้อง');
    if (iconRequired && !isButtonSectionHttpUrl_(iconUrl)) {
      throw new Error('กรุณาระบุ URL หรืออัปโหลดรูปไอคอนของปุ่ม "' + name + '"');
    }

    return {
      name: name,
      iconUrl: iconUrl,
      url: url,
      buttonColor: buttonColor,
      textColor: textColor,
      order: index + 1,
      visible: visible
    };
  });

  const sheet = getButtonSectionSheet_();
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const clearRows = Math.max(1, sheet.getMaxRows() - 1);
    sheet.getRange(2, 22, clearRows, 7).clearContent(); // V2:AB
    if (items.length) {
      sheet.getRange(2, 22, items.length, 7).setValues(items.map(function(item) {
        return [
          item.name,
          item.iconUrl,
          item.url,
          item.buttonColor,
          item.textColor,
          item.order,
          item.visible
        ];
      }));
    }
    sheet.getRange('AC2').setValue(style);
    SpreadsheetApp.flush();
  } finally {
    lock.releaseLock();
  }

  try { CacheService.getScriptCache().remove(BUTTON_SECTION_CACHE_KEY); } catch (ignore) {}
  return { style: style, items: items };
}

function uploadButtonSectionIcon_(data) {
  data = data || {};

  if (!/^data:image\/[a-zA-Z0-9.+-]+;base64,/i.test(imageData)) {

  );

}

  const defaults = getDefaultSectionLayout_();
  const defaultMap = {};
  defaults.forEach(function(item) { defaultMap[item.id] = item; });

  const out = [];
  const seen = {};
  const input = Array.isArray(items) ? items : [];

  input.forEach(function(raw) {
    raw = raw || {};
    let id = String(raw.id || '').trim();
    if (id === 'activityBox') id = 'bestPracticeBox';
    if (!id || seen[id]) return;

    const builtIn = !!defaultMap[id];
    if (!builtIn && !/^custom-section-[a-z0-9-]+$/i.test(id)) return;

    const kind = builtIn ? 'builtin' : 'custom';
    const base = builtIn ? defaultMap[id] : null;
    const title = String(raw.title || (base && base.title) || 'SECTION ใหม่').trim().slice(0, 120) || 'SECTION ใหม่';
    const rawType = String(raw.sourceType || '').trim().toLowerCase();
    const sourceType = builtIn ? '' : (rawType === 'embed' ? 'embed' : (rawType === 'image' ? 'image' : 'url'));
    const source = builtIn ? '' : String(raw.source || '');
    const detailUrl = builtIn ? '' : String(raw.detailUrl || '').trim();
    const buttonLabel = builtIn ? '' : String(raw.buttonLabel || '').trim().slice(0, 60);

    if (!builtIn) {
      if (!source.trim()) throw new Error('SECTION "' + title + '" ยังไม่ได้ระบุ URL หน้าเพจ/รูปภาพ หรือโค้ดฝัง');
      if (source.length > 45000) throw new Error('ข้อมูลของ SECTION "' + title + '" ยาวเกิน 45,000 ตัวอักษร');
      if ((sourceType === 'url' || sourceType === 'image') && !isSafeSectionUrl_(source)) {
        throw new Error('URL หน้าเพจ/รูปภาพของ SECTION "' + title + '" ไม่ถูกต้อง');
      }
      if (detailUrl && !isSafeSectionUrl_(detailUrl)) {
        throw new Error('URL รายละเอียดของ SECTION "' + title + '" ไม่ถูกต้อง');
      }
      if (detailUrl && !buttonLabel) throw new Error('กรุณากำหนดชื่อปุ่มของ SECTION "' + title + '"');
      if (buttonLabel && !detailUrl) throw new Error('กรุณาระบุ URL รายละเอียดของ SECTION "' + title + '"');
    }

    seen[id] = true;
    out.push({
      id: id,
      kind: kind,
      title: title,
      sourceType: sourceType,
      source: source,
      visible: raw.visible !== false && String(raw.visible).toLowerCase() !== 'false',
      menuLabel: '',
      height: builtIn ? 0 : normalizeSectionHeight_(raw.height),
      detailUrl: detailUrl,
      buttonLabel: buttonLabel
    });
  });

  defaults.forEach(function(base) {
    if (seen[base.id]) return;
    out.push(Object.assign({}, base));
  });

  // อ่านหนังสือสะสมเวลา: บังคับให้อยู่ก่อน cliproomBox แม้ section_layout เดิมจะบันทึกลำดับเก่าไว้
  const readBookIndex = out.findIndex(function(item) { return item.id === 'readBookTimeBox'; });
  if (readBookIndex >= 0) {
    const readBookItem = out.splice(readBookIndex, 1)[0];
    const cliproomIndex = out.findIndex(function(item) { return item.id === 'cliproomBox'; });
    if (cliproomIndex >= 0) out.splice(cliproomIndex, 0, readBookItem);
    else out.push(readBookItem);
  }

  if (out.filter(function(item) { return item.kind === 'custom'; }).length > 30) {
    throw new Error('เพิ่ม SECTION แบบกำหนดเองได้สูงสุด 30 SECTION');
  }
  return out;
}

function getLegacySectionLayout_() {
  try {
    const raw = PropertiesService.getScriptProperties().getProperty(INDEX_SECTION_LAYOUT_LEGACY_KEY);
    if (!raw) return null;
    const oldItems = JSON.parse(raw);
    if (!Array.isArray(oldItems) || !oldItems.length) return null;
    return normalizeSectionLayoutItems_(oldItems);
  } catch (err) {
    console.warn('อ่าน SECTION layout รุ่นเก่าไม่สำเร็จ: ' + err.message);
    return null;
  }
}

function getSectionLayout_() {
  const sheet = getSectionLayoutSheet_();
  const lastRow = sheet.getLastRow();

  if (lastRow < 2) {
    const legacy = getLegacySectionLayout_();
    return saveSectionLayout_({ items: legacy || getDefaultSectionLayout_() }).items;
  }

  const rows = sheet.getRange(2, 1, lastRow - 1, 12).getDisplayValues();
  const items = rows.map(function(row, index) {
    return {
      id: String(row[0] || '').trim(),
      kind: String(row[1] || '').trim(),
      title: String(row[2] || '').trim(),
      sourceType: String(row[3] || '').trim(),
      source: String(row[4] || ''),
      visible: String(row[5] || '').trim().toLowerCase() !== 'false',
      order: Number(row[6]) || (index + 1),
      menuLabel: String(row[7] || '').trim(),
      height: Number(row[8]) || 620,
      detailUrl: String(row[10] || '').trim(),
      buttonLabel: String(row[11] || '').trim()
    };
  }).filter(function(item) { return item.id; });

  items.sort(function(a, b) { return a.order - b.order; });
  const normalized = normalizeSectionLayoutItems_(items);
  if (normalized.length !== items.length) {
    return saveSectionLayout_({ items: normalized }).items;
  }
  return normalized;
}

function saveSectionLayout_(data) {
  data = data || {};
  if (Array.isArray(data)) data = { items: data };
  const items = normalizeSectionLayoutItems_(data.items);
  const sheet = getSectionLayoutSheet_();
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);

  try {
    const lastRow = sheet.getLastRow();
    if (lastRow >= 2) sheet.getRange(2, 1, lastRow - 1, 12).clearContent();

    if (items.length) {
      const now = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || 'Asia/Bangkok', 'yyyy-MM-dd HH:mm:ss');
      const values = items.map(function(item, index) {
        return [
          item.id,
          item.kind,
          item.title,
          item.sourceType || '',
          item.source || '',
          item.visible !== false,
          index + 1,
          item.menuLabel || '',
          item.kind === 'custom' ? normalizeSectionHeight_(item.height) : '',
          now,
          item.detailUrl || '',
          item.buttonLabel || ''
        ];
      });
      sheet.getRange(2, 1, values.length, 12).setValues(values);
    }
    SpreadsheetApp.flush();
    clearHomeFastCache_();
  } finally {
    lock.releaseLock();
  }
  return { items: items };
}

function upsertCustomSection_(data) {
  data = data || {};
  const current = getSectionLayout_();
  let id = String(data.id || '').trim();
  if (id && !/^custom-section-[a-z0-9-]+$/i.test(id)) throw new Error('รหัส SECTION ไม่ถูกต้อง');
  if (!id) id = 'custom-section-' + Utilities.getUuid().replace(/-/g, '').slice(0, 16).toLowerCase();

  const rawType = String(data.sourceType || '').trim().toLowerCase();
  const next = {
    id: id,
    kind: 'custom',
    title: String(data.title || '').trim(),
    sourceType: rawType === 'embed' ? 'embed' : (rawType === 'image' ? 'image' : 'url'),
    source: String(data.source || ''),
    visible: data.visible !== false,
    height: normalizeSectionHeight_(data.height),
    detailUrl: String(data.detailUrl || '').trim(),
    buttonLabel: String(data.buttonLabel || '').trim()
  };
  const index = current.findIndex(function(item) { return item.id === id; });
  if (index >= 0) current[index] = next;
  else current.push(next);
  return saveSectionLayout_({ items: current });
}

function deleteCustomSection_(data) {
  const id = String((data && data.id) || '').trim();
  if (!/^custom-section-[a-z0-9-]+$/i.test(id)) throw new Error('ลบได้เฉพาะ SECTION ที่ผู้ดูแลสร้างเพิ่มเท่านั้น');
  const current = getSectionLayout_();
  const filtered = current.filter(function(item) { return item.id !== id; });
  if (filtered.length === current.length) throw new Error('ไม่พบ SECTION ที่ต้องการลบ');
  return saveSectionLayout_({ items: filtered });
}

function uploadSectionImage_(data) {
  data = data || {};
  const imageData = String(data.imageData || '').trim();
  if (!/^data:image\/[a-zA-Z0-9.+-]+;base64,/i.test(imageData)) {
    throw new Error('กรุณาอัปโหลดไฟล์รูปภาพเท่านั้น');
  }
  const url = uploadContentAdminFile_(imageData, String(data.imageName || 'section-image'), 'dynamic-section-image', true);
  if (!url) throw new Error('อัปโหลดรูปภาพ SECTION ไม่สำเร็จ');
  return { url: url };
}

function runSectionLayoutAdminAction_(action, data) {
  const cmd = String(action || '').trim().toLowerCase();
  if (cmd === 'list') return { items: getSectionLayout_() };
  if (cmd === 'save') return saveSectionLayout_(data || {});
  if (cmd === 'upsert' || cmd === 'add' || cmd === 'update') return upsertCustomSection_(data || {});
  if (cmd === 'delete') return deleteCustomSection_(data || {});
  if (cmd === 'uploadimage') return uploadSectionImage_(data || {});
  throw new Error('ไม่รองรับคำสั่ง SECTION: ' + cmd);
}

function getHomeSummaryData_() {
  // ตัวนับเว็บไซต์ยังเพิ่มทุกครั้ง แต่ข้อมูลหนักอื่น cache ไว้
  const websiteVisitCount = incrementWebsiteVisitCount_();
  const cache = CacheService.getScriptCache();
  const cacheKey = 'HOME_SUMMARY_COUNTS_V3';
  const cached = cache.get(cacheKey);

  if (cached) {
    try {
      const data = JSON.parse(cached);
      data.websiteVisitCount = websiteVisitCount;
      return data;
    } catch (ignore) {}
  }

  const ssMain = SpreadsheetApp.openById(DATABSE_MAIN_TEATHER_ID);
  const shUser = ssMain.getSheetByName('user');

  let userTotal = 0, userPrimary = 0, userMiddle = 0, userHigh = 0;
  if (shUser && shUser.getLastRow() >= 2) {
    shUser.getRange(2, 9, shUser.getLastRow() - 1, 4).getValues().forEach(function(row) {
      userPrimary += Number(row[0]) || 0;
      userMiddle += Number(row[1]) || 0;
      userHigh += Number(row[2]) || 0;
      userTotal += Number(row[3]) || 0;
    });
  }

  let adminTotal = 0, adminPrimary = 0, adminMiddle = 0, adminHigh = 0;
  const settingSheet = ssMain.getSheetByName('Sheet');
  const studentSheetValue = settingSheet
    ? String(settingSheet.getRange('D12').getValue() || '').trim()
    : '';
  const studentSheetMatch = studentSheetValue.match(/\/d\/([a-zA-Z0-9-_]+)/);
  const studentSheetId = studentSheetMatch ? studentSheetMatch[1] : studentSheetValue;

  if (studentSheetId) {
    const ssStudent = SpreadsheetApp.openById(studentSheetId);
    const shStudent = ssStudent.getSheetByName('รายชื่อนักศึกษาทั้งหมด');
    if (shStudent && shStudent.getLastRow() >= 2) {
      shStudent.getRange(2, 2, shStudent.getLastRow() - 1, 3).getValues().forEach(function(row) {
        const studentId = String(row[0] || '').trim();
        const level = String(row[2] || '').trim();
        if (!studentId) return;
        if (level === 'ประถม') adminPrimary++;
        else if (level === 'ม.ต้น') adminMiddle++;
        else if (level === 'ม.ปลาย') adminHigh++;
      });
      adminTotal = adminPrimary + adminMiddle + adminHigh;
    }
  }

  let adminLoginCount = 0, userLoginCount = 0;
  const shLog = ssMain.getSheetByName('Log');
  if (shLog && shLog.getLastRow() >= 2) {
    shLog.getRange(2, 5, shLog.getLastRow() - 1, 1).getDisplayValues().forEach(function(row) {
      const loginType = String(row[0] || '').trim();
      if (!loginType) return;
      if (loginType.toLowerCase() === 'admin') adminLoginCount++;
      else userLoginCount++;
    });
  }

  let registrationCount = 0;
  const learningBaseSS = SpreadsheetApp.openById(DB_ID);
  const studentsSheet = learningBaseSS.getSheetByName('students');
  if (studentsSheet && studentsSheet.getLastRow() >= 2) {
    registrationCount = studentsSheet
      .getRange(2, 1, studentsSheet.getLastRow() - 1, 1)
      .getDisplayValues()
      .reduce(function(count, row) {
        return count + (String(row[0] || '').trim() ? 1 : 0);
      }, 0);
  }

  const data = {
    userTotal: userTotal,
    userPrimary: userPrimary,
    userMiddle: userMiddle,
    userHigh: userHigh,
    adminTotal: adminTotal,
    adminPrimary: adminPrimary,
    adminMiddle: adminMiddle,
    adminHigh: adminHigh,
    registrationCount: registrationCount,
    loginCount: adminLoginCount + userLoginCount,
    adminLoginCount: adminLoginCount,
    userLoginCount: userLoginCount,
    websiteVisitCount: websiteVisitCount
  };

  try {
    const cacheData = Object.assign({}, data);
    delete cacheData.websiteVisitCount;
    cache.put(cacheKey, JSON.stringify(cacheData), 120);
  } catch (ignore) {}

  return data;
}

/*
รีเซต ค่าเป็น 0 เลือกฟังก์ชัน resetWebsiteVisitCount กดปุ่ม Run
*/
function resetWebsiteVisitCount() {
  PropertiesService
    .getScriptProperties()
    .setProperty('WEBSITE_VISIT_COUNT', '0');

  console.log('รีเซตจำนวนการเข้าเว็บไซต์เป็น 0 แล้ว');
}
function incrementWebsiteVisitCount_() {
  const propertyName = 'WEBSITE_VISIT_COUNT';
  const lock = LockService.getScriptLock();

  try {
    lock.waitLock(10000);

    const properties = PropertiesService.getScriptProperties();
    const currentCount = Number(properties.getProperty(propertyName)) || 0;
    const nextCount = currentCount + 1;

    properties.setProperty(propertyName, String(nextCount));
    return nextCount;
  } catch (err) {
    console.error('incrementWebsiteVisitCount_ error:', err);
    return Number(
      PropertiesService.getScriptProperties().getProperty(propertyName)
    ) || 0;
  } finally {
    if (lock.hasLock()) lock.releaseLock();
  }
}

    };
  }).filter(function(item) {
    return item.title || item.detail;
  });

  // fallback ข้อมูลเดิม เพื่อไม่กระทบระบบเก่า หาก P:Q ยังไม่มีข้อมูล
  function findAboutValue_(labels) {
    const targets = labels.map(function(label) {
      return String(label).replace(/\s+/g, '');
    });

    for (let rowIndex = 0; rowIndex < legacyVisionRange.length; rowIndex++) {
      const row = legacyVisionRange[rowIndex];
      for (let colIndex = 0; colIndex < row.length; colIndex++) {
        const label = String(row[colIndex] || '').trim().replace(/\s+/g, '');
        if (targets.indexOf(label) === -1) continue;

        for (let valueIndex = colIndex + 1; valueIndex < row.length; valueIndex++) {
          const value = String(row[valueIndex] || '').trim();
          if (value) return value;
        }
      }
    }
    return '';
  }

  const legacyVision = findAboutValue_(['วิสัยทัศน์']);
  const legacyIdentity = findAboutValue_(['อัตลักษณ์', 'เอกลักษณ์']);
  const legacyMission = findAboutValue_(['พันธกิจ']);

  const sections = visionSections.length ? visionSections : [
    { title: 'วิสัยทัศน์', detail: legacyVision },
    { title: 'อัตลักษณ์', detail: legacyIdentity },
    { title: 'พันธกิจ', detail: legacyMission }
  ].filter(function(item) {
    return item.detail;
  });

  return {
    contact: {
      organization: String(contactValues[0][0] || '').trim(),
      announcementText: centralAnnouncement.text,
        announcementUrl: centralAnnouncement.url,
      coordinate: String(contactValues[1][0] || '').trim(),
      address: String(contactValues[2][0] || '').trim(),
      phone: String(contactValues[3][0] || '').trim(),
      facebook: String(contactValues[4][0] || '').trim(),
      line: String(contactValues[5][0] || '').trim(),
      youtube: String((contactValues[6] && contactValues[6][0]) || '').trim()
    },
    vision: {
      kicker: visionKicker || 'VISION, IDENTITY, MISSION',
      title: visionTitle || 'วิสัยทัศน์ อัตลักษณ์ พันธกิจ',
      sections: sections,
      vision: legacyVision,
      identity: legacyIdentity,
      mission: legacyMission
    },
    hero: {
      kicker: String(heroValues[0][0] || '').trim(),
      title: String(heroValues[1][0] || '').trim(),
      description: String(heroValues[2][0] || '').trim()
    },
    footer: {
      organization: String(contactValues[0][0] || '').trim(),
      address: String(contactValues[2][0] || '').trim(),
      phone: String(contactValues[3][0] || '').trim(),
      description: String(heroValues[1][0] || '').trim()
    }
  };
}



// ============================================================
// Website image upload (Admin -> Google Drive -> website_image)
// Folder ID is read from Sheet!D11 in DATABSE_MAIN_TEATHER_ID.
// Supports either a raw folder ID or a Google Drive folder URL.
// ============================================================
function getWebsiteImageFolderId_() {
  const ss = SpreadsheetApp.openById(DATABSE_MAIN_TEATHER_ID);
  const sheet = ss.getSheetByName('Sheet');
  if (!sheet) throw new Error('ไม่พบชีต Sheet สำหรับอ่าน Folder ID รูปเว็บไซต์');

  const raw = String(sheet.getRange('D11').getDisplayValue() || '').trim();
  if (!raw) throw new Error('ไม่พบ Folder ID รูปเว็บไซต์ที่ Sheet!D11');

  const match = raw.match(/[-\w]{25,}/);
  const folderId = match ? match[0] : raw;
  if (!folderId) throw new Error('Folder ID รูปเว็บไซต์ไม่ถูกต้อง');
  return folderId;
}

function uploadWebsiteImage_(dataUrl, filename, prefix) {
  const value = String(dataUrl || '').trim();
  if (!value) return '';

  const match = value.match(/^data:(image\/[a-zA-Z0-9.+-]+);base64,(.+)$/);
  if (!match) throw new Error('รูปภาพมีรูปแบบไม่ถูกต้อง');

  const contentType = match[1];
  const bytes = Utilities.base64Decode(match[2]);
  if (bytes.length > 3 * 1024 * 1024) {
    throw new Error('รูปภาพหลังย่อมีขนาดใหญ่เกิน 3 MB');
  }

  const folderId = getWebsiteImageFolderId_();
  const safeBase = String(filename || (prefix || 'website-image') + '.jpg')
    .replace(/[\\\\/:*?"<>|]/g, '_')
    .replace(/^\.+/, '') || 'website-image.jpg';
  const safeName = String(prefix || 'website') + '-' + Date.now() + '-' + safeBase;
  const blob = Utilities.newBlob(bytes, contentType, safeName);

  let file;
  try {
    file = Drive.Files.create(
      {
        name: safeName,
        parents: [folderId]
      },
      blob,
      {
        supportsAllDrives: true,
        fields: 'id,name'
      }
    );
  } catch (err) {
    throw new Error('อัปโหลดรูปเข้า Google Drive ไม่สำเร็จ: ' + (err && err.message ? err.message : err));
  }

  try {
    Drive.Permissions.create(
      { role: 'reader', type: 'anyone' },
      file.id,
      { supportsAllDrives: true }
    );
  } catch (err) {
    // Delete the unusable private file so a failed save does not leave junk behind.
    try { Drive.Files.remove(file.id); } catch (ignore) {}
    throw new Error('ตั้งสิทธิ์รูปภาพเป็นสาธารณะไม่สำเร็จ: ' + (err && err.message ? err.message : err));
  }

  return 'https://lh3.googleusercontent.com/d/' + file.id;
}

function getEditWebsiteData_(editor) {
  const ss = SpreadsheetApp.openById(DATABSE_MAIN_TEATHER_ID);
  const type = String(editor || '').trim().toLowerCase();
  let sheet, labels, values, headers;

  if (type === 'text') {
    sheet = ss.getSheetByName('setting');
    if (!sheet) throw new Error('ไม่พบชีต setting');
    headers = sheet.getRange('S1:T1').getDisplayValues()[0];
    labels = sheet.getRange('S2:S4').getDisplayValues().map(function (row) { return row[0]; });
    values = sheet.getRange('T2:T4').getDisplayValues().map(function (row) { return row[0]; });
  } else if (type === 'image') {
    sheet = ss.getSheetByName(WEBSITE_IMAGE_SHEET);
    if (!sheet) throw new Error('ไม่พบชีต website_image');
    headers = sheet.getRange('A1:B1').getDisplayValues()[0];
    labels = sheet.getRange('A2:A4').getDisplayValues().map(function (row) { return row[0]; });
    values = sheet.getRange('B2:B4').getDisplayValues().map(function (row) { return row[0]; });
  } else if (type === 'contact') {
    sheet = ss.getSheetByName('setting');
    if (!sheet) throw new Error('ไม่พบชีต setting');

    // ผู้ดูแล: แสดงข้อมูล M2:N8 เท่านั้น
    // ตำแหน่งในชีตคงเดิม: N3 = พิกัด, N4 = ที่อยู่, N8 = YouTube
    headers = sheet.getRange('M1:N1').getDisplayValues()[0];
    labels = sheet.getRange('M2:M8').getDisplayValues().map(function (row) { return row[0]; });
    values = sheet.getRange('N2:N8').getDisplayValues().map(function (row) { return row[0]; });

    // หาก M8 ว่าง ให้ popup ยังแสดงชื่อ Youtube
    if (!String(labels[6] || '').trim()) labels[6] = 'Youtube';
  } else if (type === 'vision') {
    sheet = ss.getSheetByName('setting');
    if (!sheet) throw new Error('ไม่พบชีต setting');
    headers = sheet.getRange('P1:Q1').getDisplayValues()[0];
    const lastRow = Math.max(3, sheet.getLastRow());
    const visionRows = sheet.getRange(2, 16, lastRow - 1, 2).getDisplayValues(); // P2:Q

    let keep = Math.max(2, visionRows.length);
    while (keep > 2) {
      const row = visionRows[keep - 1] || ['', ''];
      if (String(row[0] || '').trim() || String(row[1] || '').trim()) break;
      keep--;
    }

    return {
      success: true,
      editor: type,
      headers: headers.map(function(value) { return String(value || ''); }),
      rows: visionRows.slice(0, keep).map(function(row) {
        return {
          title: String(row[0] || ''),
          detail: String(row[1] || '')
        };
      })
    };
  } else {
    throw new Error('ประเภทตารางไม่ถูกต้อง');
  }

  return {
    success: true,
    editor: type,
    headers: headers.map(function (value) { return String(value || ''); }),
    rows: labels.map(function (label, index) {
      return { label: String(label || ''), value: String(values[index] || '') };
    })
  };
}

function saveEditWebsiteData_(editor, valuesJson, uploadData) {
  const ss = SpreadsheetApp.openById(DATABSE_MAIN_TEATHER_ID);
  const type = String(editor || '').trim().toLowerCase();
  let values;
  try {
    values = Array.isArray(valuesJson) ? valuesJson.slice() : JSON.parse(String(valuesJson || '[]'));
  } catch (err) {
    throw new Error('รูปแบบข้อมูลไม่ถูกต้อง');
  }
  if (!Array.isArray(values)) throw new Error('รูปแบบข้อมูลไม่ถูกต้อง');

  const uploads = {};

  // Upload first, then hold the spreadsheet lock only while writing cells.
  // This makes the popup faster and prevents the selected image from being discarded.
  if (type === 'image') {
    if (values.length !== 3) throw new Error('ข้อมูลโลโก้ ชื่อ และรูปหัวเว็บไซต์ต้องมี 3 รายการ');

    const data = uploadData && typeof uploadData === 'object' ? uploadData : {};

    if (String(data.logoData || '').trim()) {
      values[0] = uploadWebsiteImage_(data.logoData, data.logoName, 'website-logo');
      uploads.logoUrl = values[0];
    }

    if (String(data.heroData || '').trim()) {
      values[2] = uploadWebsiteImage_(data.heroData, data.heroName, 'website-hero');
      uploads.heroUrl = values[2];
    }

    values[0] = String(values[0] || '').trim();
    values[1] = String(values[1] || '').trim();
    values[2] = String(values[2] || '').trim();

    if (!values[0]) throw new Error('กรุณาอัปโหลดหรือระบุ URL โลโก้');
    if (!values[2]) throw new Error('กรุณาอัปโหลดหรือระบุ URL รูปหัวเว็บไซต์');
  }

  if (type === 'contact') {
    if (values.length !== 7) {
      throw new Error('ข้อมูลติดต่อ M2:N8 ต้องมี 7 รายการ');
    }

    values = values.map(function(value) {
      return String(value || '').trim();
    });

    const coordinate = values[1]; // N3
    if (coordinate) {
      const match = coordinate.match(/^\s*(-?(?:\d+(?:\.\d+)?|\.\d+))\s*,\s*(-?(?:\d+(?:\.\d+)?|\.\d+))\s*$/);
      if (!match) {
        throw new Error('พิกัด N3 ไม่ถูกต้อง กรุณาระบุเป็น ละติจูด, ลองจิจูด');
      }
      const lat = Number(match[1]);
      const lng = Number(match[2]);
      if (!isFinite(lat) || !isFinite(lng) || lat < -90 || lat > 90 || lng < -180 || lng > 180) {
        throw new Error('พิกัด N3 อยู่นอกช่วงที่ถูกต้อง');
      }
    }

    const phone = values[3]; // N5
    if (phone && !/^\d{6,15}$/.test(phone)) {
      throw new Error('เบอร์โทร N5 ต้องเป็นตัวเลข 6-15 หลัก และสามารถขึ้นต้นด้วย 0 ได้');
    }

    const facebook = values[4]; // N6
    if (facebook && !/^https?:\/\/(?:[^\/]+\.)?facebook\.com(?:\/|$)/i.test(facebook)) {
      throw new Error('N6 ต้องเป็น URL ของ Facebook เท่านั้น');
    }

    const line = values[5]; // N7
    if (line && !/^https?:\/\/(?:[^\/]+\.)?(?:line\.me|lin\.ee)(?:\/|$)/i.test(line)) {
      throw new Error('N7 ต้องเป็น URL ของ Line เท่านั้น');
    }

    const youtube = values[6]; // N8
    if (youtube && !/^https?:\/\/(?:[^\/]+\.)?(?:youtube\.com|youtu\.be)(?:\/|$)/i.test(youtube)) {
      throw new Error('N8 ต้องเป็น URL ของ YouTube เท่านั้น');
    }
  }


  if (type === 'vision') {
    if (values.length < 2) throw new Error('ข้อมูล Vision ต้องมีอย่างน้อย P2:Q3');
    if (values.length > 100) throw new Error('เพิ่มข้อมูลได้สูงสุด 100 แถวต่อครั้ง');

    values = values.map(function(row) {
      if (!Array.isArray(row)) return ['', ''];
      return [String(row[0] || '').trim(), String(row[1] || '').trim()];
    });
  }
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    if (type === 'text') {
      if (values.length !== 3) throw new Error('ข้อมูลข้อความต้องมี 3 รายการ');
      const sheet = ss.getSheetByName('setting');
      if (!sheet) throw new Error('ไม่พบชีต setting');
      sheet.getRange('T2:T4').setValues(values.map(function (value) { return [String(value || '')]; }));
    } else if (type === 'image') {
      const sheet = ss.getSheetByName(WEBSITE_IMAGE_SHEET);
      if (!sheet) throw new Error('ไม่พบชีต website_image');
      sheet.getRange('B2:B4').setValues(values.map(function (value) { return [String(value || '')]; }));
    } else if (type === 'contact') {
      const sheet = ss.getSheetByName('setting');
      if (!sheet) throw new Error('ไม่พบชีต setting');
      // บันทึกเฉพาะ N2:N8 = 7 ค่า
      const range = sheet.getRange('N2:N8');
      range.setNumberFormat('@');
      range.setValues(values.map(function (value) { return [String(value || '')]; }));
    } else if (type === 'vision') {
      const sheet = ss.getSheetByName('setting');
      if (!sheet) throw new Error('ไม่พบชีต setting');

      const lastSheetRow = Math.max(3, sheet.getLastRow());
      const existing = sheet.getRange(2, 16, lastSheetRow - 1, 2).getDisplayValues();
      let existingCount = 2;
      for (let i = existing.length - 1; i >= 0; i--) {
        if (String(existing[i][0] || '').trim() || String(existing[i][1] || '').trim()) {
          existingCount = Math.max(2, i + 1);
          break;
        }
      }

      const clearCount = Math.max(existingCount, values.length);
      sheet.getRange(2, 16, clearCount, 2).clearContent();
      const range = sheet.getRange(2, 16, values.length, 2);
      range.setNumberFormat('@');
      range.setValues(values);
    } else {
      throw new Error('ประเภทตารางไม่ถูกต้อง');
    }
    SpreadsheetApp.flush();
    clearHomeFastCache_();
  } finally {
    lock.releaseLock();
  }

  return {
    success: true,
    editor: type,
    values: type === 'vision'
      ? values.map(function(row) { return [String(row[0] || ''), String(row[1] || '')]; })
      : values.map(function (value) { return String(value || ''); }),
    uploads: uploads
  };
}

function doPost(e) {
  try {
    const content =
      e &&
      e.postData &&
      e.postData.contents
        ? e.postData.contents
        : '{}';

    const body = JSON.parse(content);

    const mode = String(body.mode || '').trim();
    const action = String(body.action || '').trim();
    const data = body.data || {};

    if (mode === 'adminlogin') {
      return jsonOutput_(adminLogin_(body.username, body.password));
    }

    if (mode === 'adminforgot') {
      return jsonOutput_(sendAdminCredentials_(body.email));
    }

    if (mode === 'adminstorage') {
      requireAdminSession_(body.token);
      return jsonOutput_({
        success: true,
        data: getAdminStorageUsage_(String(body.fresh || '') === '1')
      });
    }

    if (mode === 'sectionlayoutadmin') {
      requireAdminSession_(body.token);
      return jsonOutput_({
        success: true,
        data: runSectionLayoutAdminAction_(action, body.data || {})
      });
    }

    if (mode === 'templatesettingadmin') {
      requireAdminSession_(body.token);
      return jsonOutput_({
        success: true,
        data: saveTemplateSetting_(body.template || (body.data && body.data.template))
      });
    }


    if (mode === 'templatecolorsettingadmin') {
      requireAdminSession_(body.token);
      return jsonOutput_({ success: true, data: saveTemplateColorSetting_(body.template, body.palette) });
    }

    if (mode === 'herocontentadmin') {
      requireAdminSession_(body.token);
      return jsonOutput_({
        success: true,
        data: saveHeroContentSetting_(body.template, body.data || body)
      });
    }

    if (mode === 'editwebsite') {
      requireAdminSession_(body.token);
      return jsonOutput_(getEditWebsiteData_(body.editor));
    }

    if (mode === 'saveeditwebsite') {
      requireAdminSession_(body.token);
      return jsonOutput_(saveEditWebsiteData_(body.editor, body.values, body.data || {}));
    }

    if (mode === 'newsadmin') {
      requireAdminSession_(body.token);
      return jsonOutput_({ success: true, data: runNewsAdminAction_(body.action, body.data || {}) });
    }


    // จัดการโพสต์ Facebook จากปุ่ม manageFacebookButton บน index.html
    if (mode === 'facebookadmin') {
      requireAdminSession_(body.token);
      return jsonOutput_({
        success: true,
        data: runFacebookAdminAction_(action, data)
      });
    }


    // จัดการ buttonsection: รายการปุ่ม/รูปแบบ/อัปโหลดไอคอน
    if (mode === 'buttonsectionadmin') {
      requireAdminSession_(body.token);
      return jsonOutput_({
        success: true,
        data: runButtonSectionAdminAction_(action, data)
      });
    }

    // จัดการรายงานผลการปฏิบัติการจาก report.html
    if (mode === 'reportadmin') {
      requireAdminSession_(body.token);
      return jsonOutput_({
        success: true,
        data: runPerformanceReportAdminAction_(action, data)
      });
    }

    const contentAdminModes = [
      'teamadmin',
      'bestpracticeadmin',
      'innovationadmin',
      'mediaadmin',
      'rewardadmin',
      'courseadmin'
    ];

    if (contentAdminModes.indexOf(mode) !== -1) {
      requireAdminSession_(body.token);
      return jsonOutput_({
        success: true,
        data: runContentAdminAction_(mode, action, data)
      });
    }

    /*
     * รองรับทั้ง JavaScript รุ่นเดิม:
     * { mode: "learning", action: "...", data: {...} }
     *
     * และรุ่นใหม่:
     * { action: "...", data: {...} }
     */
    if (!action) {
      throw new Error('ไม่พบคำสั่ง action');
    }

    if (mode && mode !== 'learning') {
      throw new Error('Invalid mode');
    }

    return jsonOutput_({
      success: true,
      data: runLearningAction_(action, data)
    });

  } catch (err) {
    return jsonOutput_({
      success: false,
      message: err.message
    });
  }
}

function validateFacebookEmbedCode_(embedCode) {
  const code = String(embedCode || '').trim();
  if (!code) return false;

  // ต้องเป็น iframe เพียงรายการเดียว และ src ต้องเป็น Facebook Post Plugin
  // รองรับ URL โพสต์จาก facebook.com ทุก subdomain เช่น www., web., m., mbasic.
  if (!/^<iframe\b[\s\S]*<\/iframe>$/i.test(code)) return false;

  const srcMatch = code.match(/<iframe[^>]*\bsrc\s*=\s*["']([^"']+)["']/i);
  if (!srcMatch) return false;

  const embedUrl = String(srcMatch[1] || '').replace(/&(?:amp|#38|#x26);/gi, '&').trim();
  if (!/^https:\/\/(?:[a-z0-9-]+\.)*facebook\.com\/plugins\/post\.php\?/i.test(embedUrl)) return false;

  const hrefMatch = embedUrl.match(/[?&]href=([^&]+)/i);
  if (!hrefMatch) return false;

  let facebookUrl = hrefMatch[1];
  try {
    facebookUrl = decodeURIComponent(facebookUrl);
  } catch (ignore) {}

  return /^https:\/\/(?:[a-z0-9-]+\.)*facebook\.com\//i.test(facebookUrl);
}

function saveFacebookAdmin_(data) {
  data = data || {};
  const posterKey = getFacebookPosterKey_();
  const embedCode = String(data.embedCode || '').trim();

  if (!validateFacebookEmbedCode_(embedCode)) {
    throw new Error('โค้ดฝังต้องเป็น iframe จาก Facebook Post Plugin เท่านั้น');
  }

  const sheet = getFacebookAdminSheet_();
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);

  try {
    const requestedRow = Number(data.rowNumber || 0);
    const editing = requestedRow >= 2 && requestedRow <= sheet.getLastRow();
    let rowNumber = requestedRow;
    let date = formatFacebookAdminDate_(new Date());

    if (editing) {
      const existingKey = String(sheet.getRange(rowNumber, 1).getDisplayValue() || '').trim();
      if (existingKey !== posterKey) {
        throw new Error('ไม่อนุญาตให้แก้ไขโพสต์ Facebook ของคีย์อื่น');
      }
      date = String(sheet.getRange(rowNumber, 3).getDisplayValue() || '').trim() || date;

      // ผู้โพสต์ใช้คีย์จากการตั้งค่าของเว็บไซต์เท่านั้น ไม่รับค่าที่แก้จาก Browser
      sheet.getRange(rowNumber, 1, 1, 3).setValues([[posterKey, embedCode, date]]);
    } else {
      // ใช้ appendRow เพื่อให้การเพิ่มจากหลายเว็บไซต์ลงชีตกลางชนกันยากที่สุด
      sheet.appendRow([posterKey, embedCode, date]);
      rowNumber = sheet.getLastRow();
    }

    SpreadsheetApp.flush();
    clearFacebookItemsCache_();

    return {
      rowNumber: rowNumber,
      item: {
        rowNumber: rowNumber,
        poster: posterKey,
        embedCode: embedCode,
        date: date
      },
      posterKey: posterKey
    };
  } finally {
    lock.releaseLock();
  }
}

function deleteFacebookAdmin_(rowNumber) {
  const posterKey = getFacebookPosterKey_();
  const sheet = getFacebookAdminSheet_();
  const row = Number(rowNumber || 0);

  if (row < 2 || row > sheet.getLastRow()) {
    throw new Error('ไม่พบโพสต์ Facebook ที่ต้องการลบ');
  }

  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    if (row > sheet.getLastRow()) {
      throw new Error('ไม่พบโพสต์ Facebook ที่ต้องการลบ');
    }

    const existingKey = String(sheet.getRange(row, 1).getDisplayValue() || '').trim();
    if (existingKey !== posterKey) {
      throw new Error('ไม่อนุญาตให้ลบโพสต์ Facebook ของคีย์อื่น');
    }

    sheet.deleteRow(row);
    SpreadsheetApp.flush();
    clearFacebookItemsCache_();
    return { deleted: true, posterKey: posterKey };
  } finally {
    lock.releaseLock();
  }
}

function clearFacebookItemsCache_() {
  try {
    CacheService.getScriptCache().remove('FACEBOOK_ITEMS_V1');
  } catch (ignore) {}
}

function getContentAdminOptions_(mode) {
  const type = String(mode || '').trim().toLowerCase();
  const optionColumns = {
    innovationadmin: 32, // AF2:AF
    mediaadmin: 34,      // AH2:AH
    courseadmin: 36      // AJ2:AJ
  };
  const column = optionColumns[type];
  if (!column) return [];

  const ss = SpreadsheetApp.openById(DATABSE_MAIN_ID);
  const sheet = ss.getSheetByName('setting');
  if (!sheet || sheet.getLastRow() < 2) return [];

  const values = sheet
    .getRange(2, column, sheet.getLastRow() - 1, 1)
    .getDisplayValues();

  const seen = {};
  const options = [];
  values.forEach(function(row) {
    const value = String((row && row[0]) || '').trim();
    if (!value || seen[value]) return;
    seen[value] = true;
    options.push(value);
  });
  return options;
}

  try {
    const ss = SpreadsheetApp.openById(DATABSE_MAIN_ID);
    const sheet = ss.getSheetByName('Sheet');

    return String(sheet.getRange('D23').getDisplayValue() || '').trim();

  }
}

function isSharedContentAdminMode_(mode) {
  return [
    'bestpracticeadmin',
    'innovationadmin',
    'mediaadmin',
    'rewardadmin',
    'courseadmin'
  ].indexOf(String(mode || '').trim().toLowerCase()) !== -1;
}

function ensureSharedContentKeyColumn_(sheet, config) {
  if (!sheet || !config || !config.shared || !config.keyColumn) return;

  const keyColumn = Number(config.keyColumn);
  if (sheet.getMaxColumns() < keyColumn) {
    sheet.insertColumnsAfter(
      sheet.getMaxColumns(),
      keyColumn - sheet.getMaxColumns()
    );
  }

  const keyHeader = 'คีย์ผู้โพสต์';
  if (String(sheet.getRange(1, keyColumn).getDisplayValue() || '').trim() !== keyHeader) {
    sheet.getRange(1, keyColumn).setValue(keyHeader);
  }

  // ข้อมูลเดิมในฐานกลางถือเป็นข้อมูลของเว็บไซต์อำเภอ
  // เติมเฉพาะแถวที่มีข้อมูลเดิมแต่ยังไม่มีคีย์ โดยไม่แตะคีย์ที่กำหนดไว้แล้ว
  const lastRow = sheet.getLastRow();
  const districtKey = getDistrictContentPosterKey_();
  if (lastRow < 2 || !districtKey) return;

  const dataWidth = Number(config.baseColumns || Math.max(1, keyColumn - 1));
  const dataRows = sheet.getRange(2, 1, lastRow - 1, dataWidth).getDisplayValues();
  const keyValues = sheet.getRange(2, keyColumn, lastRow - 1, 1).getDisplayValues();
  let changed = false;

  const nextKeys = keyValues.map(function(row, index) {
    const current = String((row && row[0]) || '').trim();
    if (current) return [current];

    const hasData = (dataRows[index] || []).some(function(value) {
      return String(value || '').trim() !== '';
    });

    if (hasData) {
      changed = true;
      return [districtKey];
    }
    return [''];
  });

  if (changed) {
    sheet.getRange(2, keyColumn, nextKeys.length, 1).setValues(nextKeys);
    SpreadsheetApp.flush();
  }
}

function runContentAdminAction_(mode, action, data) {
  const type = String(mode || '').trim().toLowerCase();
  const cmd = String(action || '').trim().toLowerCase();
  const payload = data || {};

  if (cmd === 'list') {
    return listContentAdminItems_(type);
  }

  if (cmd === 'key') {
    if (!isSharedContentAdminMode_(type)) return { posterKey: '' };
    return { posterKey: getContentPosterKey_() };
  }

  if (cmd === 'options') {
    return getContentAdminOptions_(type);
  }

  if (cmd === 'save') {
    return saveContentAdminItem_(type, payload);
  }

  if (cmd === 'delete') {
    return deleteContentAdminItem_(type, payload);
  }

  throw new Error('ไม่รองรับคำสั่ง action: ' + cmd);
}

function getContentAdminSheet_(mode) {
  const config = getContentAdminConfig_(mode);
  const spreadsheetId = config.shared ? DATABSE_MAIN_ID : DATABSE_MAIN_TEATHER_ID;
  const ss = SpreadsheetApp.openById(spreadsheetId);
  let sheet = ss.getSheetByName(config.sheetName);

  if (!sheet) {
    sheet = ss.insertSheet(config.sheetName);
    sheet.getRange(1, 1, 1, config.headers.length).setValues([config.headers]);
  } else if (sheet.getLastRow() === 0) {
    sheet.getRange(1, 1, 1, config.headers.length).setValues([config.headers]);
  }

  if (config.shared) {
    ensureSharedContentKeyColumn_(sheet, config);
  }

  if (mode === 'courseadmin') {
    if (sheet.getMaxColumns() < 7) {
      sheet.insertColumnsAfter(sheet.getMaxColumns(), 7 - sheet.getMaxColumns());

      sheet.getRange(1, 7).setValue('หลักสูตร');
    }
  }

  return sheet;
}

function saveContentAdminItem_(mode, data) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);

  try {
    const sheet = getContentAdminSheet_(mode);
    const config = getContentAdminConfig_(mode);
    const posterKey = config.shared ? getContentPosterKey_() : '';
    const requestedRow = Number(data.rowNumber || 0);
    const editing = requestedRow >= 2 && requestedRow <= sheet.getLastRow();
    let rowNumber = editing ? requestedRow : Math.max(2, sheet.getLastRow() + 1);
    const existing = editing
      ? sheet.getRange(rowNumber, 1, 1, config.columns).getDisplayValues()[0]
      : new Array(config.columns).fill('');

    if (config.shared && editing) {
      const existingKey = String(existing[config.keyColumn - 1] || '').trim();
      if (existingKey !== posterKey) {
        throw new Error('ไม่อนุญาตให้แก้ไขข้อมูลของคีย์อื่น');
      }
    }

    const order = editing
      ? (String(existing[0] || '').trim() || String(rowNumber - 1))
      : String(Math.max(1, rowNumber - 1));

    const today = Utilities.formatDate(
      new Date(),
      Session.getScriptTimeZone() || 'Asia/Bangkok',
      'dd/MM/yyyy'
    );

    let values;

    if (mode === 'teamadmin') {
      const name = String(data.name || '').trim();
      const position = String(data.position || '').trim();
      if (!name || !position) throw new Error('กรุณากรอกชื่อ-นามสกุลและตำแหน่ง');

      let image = String(data.imageUrl || '').trim();
      if (String(data.imageData || '').trim()) {
        image = uploadContentAdminFile_(
          data.imageData,
          data.imageName,
          'team',
          true
        );
      }
      if (!image) throw new Error('กรุณาระบุรูปภาพบุคลากร');

      values = [order, name, position, image];
    }

    if (mode === 'bestpracticeadmin') {
      let poster = String(data.posterUrl || '').trim();
      if (String(data.posterData || '').trim()) {
        poster = uploadContentAdminFile_(data.posterData, data.posterName, 'best-practice-poster', true);
      }

      let fileUrl = String(data.fileUrl || '').trim();
      if (String(data.fileData || '').trim()) {
        fileUrl = uploadContentAdminFile_(data.fileData, data.fileName, 'best-practice-file', false);
      }

      if (!poster || !fileUrl) throw new Error('กรุณาระบุภาพปกและไฟล์');
      values = [order, poster, fileUrl, String(existing[3] || '').trim() || today, posterKey];
    }

    if (mode === 'innovationadmin') {
      const mediaType = String(data.mediaType || '').trim();
      const mediaName = String(data.mediaName || '').trim();
      const producer = String(data.producer || '').trim();
      const organization = String(data.organization || '').trim();
      if (!mediaType || !mediaName || !producer || !organization) {
        throw new Error('กรุณากรอกประเภทสื่อ ชื่อสื่อ ผู้ผลิต และหน่วยงาน');
      }
      const innovationTypes = getContentAdminOptions_('innovationadmin');
      if (innovationTypes.length && innovationTypes.indexOf(mediaType) === -1) {
        throw new Error('ประเภทสื่อไม่อยู่ในรายการ setting!AF2:AF');
      }

      let poster = String(data.posterUrl || '').trim();
      if (String(data.posterData || '').trim()) {
        poster = uploadContentAdminFile_(data.posterData, data.posterName, 'innovation-poster', true);
      }

      let fileUrl = String(data.fileUrl || '').trim();
      if (String(data.fileData || '').trim()) {
        fileUrl = uploadContentAdminFile_(data.fileData, data.fileName, 'innovation-file', false);
      }

      if (!poster || !fileUrl) throw new Error('กรุณาระบุรูปภาพปกและไฟล์');
      values = [order, mediaType, mediaName, producer, organization, poster, fileUrl, posterKey];
    }

    if (mode === 'mediaadmin') {
      const mediaType = String(data.mediaType || '').trim();
      const title = String(data.title || '').trim();
      const subject = String(data.subject || '').trim();
      const grade = String(data.grade || '').trim();
      const owner = String(data.owner || '').trim();
      if (!mediaType || !title || !subject || !grade || !owner) {
        throw new Error('กรุณากรอกประเภทสื่อ เรื่อง วิชา ชั้น และเจ้าของผลงาน');
      }
      const mediaTypes = getContentAdminOptions_('mediaadmin');
      if (mediaTypes.length && mediaTypes.indexOf(mediaType) === -1) {
        throw new Error('ประเภทสื่อไม่อยู่ในรายการ setting!AH2:AH');
      }

      let fileUrl = String(data.fileUrl || '').trim();
      if (String(data.fileData || '').trim()) {
        fileUrl = uploadContentAdminFile_(data.fileData, data.fileName, 'media-file', false);
      }

      let image = String(data.imageUrl || '').trim();
      if (String(data.imageData || '').trim()) {
        image = uploadContentAdminFile_(data.imageData, data.imageName, 'media-poster', true);
      }

      if (!fileUrl || !image) throw new Error('กรุณาระบุไฟล์และรูปภาพปก');
      values = [
        order,
        mediaType,
        title,
        subject,
        grade,
        fileUrl,
        image,
        owner,
        String(existing[8] || '').trim() || today,
        posterKey
      ];
    }

    if (mode === 'rewardadmin') {
      const award = String(data.award || '').trim();
      const organization = String(data.organization || '').trim();
      const receivedDate = String(data.receivedDate || '').trim();
      const recipient = String(data.recipient || '').trim();
      if (!award || !organization || !receivedDate || !recipient) {
        throw new Error('กรุณากรอกข้อมูลรางวัลให้ครบ');
      }

      function resolveRewardImage_(key, oldValue) {
        let url = String(data[key + 'Url'] || '').trim();
        if (String(data[key + 'Data'] || '').trim()) {
          url = uploadContentAdminFile_(
            data[key + 'Data'],
            data[key + 'Name'],
            'reward-' + key,
            true
          );
        }
        return url || String(oldValue || '').trim();
      }

      const mainImage = resolveRewardImage_('mainImage', existing[5]);
      const image1 = resolveRewardImage_('image1', existing[6]);
      const image2 = resolveRewardImage_('image2', existing[7]);
      const image3 = resolveRewardImage_('image3', existing[8]);
      const detailUrl = String(data.detailUrl || '').trim();

      if (!mainImage) throw new Error('กรุณาระบุรูปภาพปกรางวัล');

      values = [
        order,
        award,
        organization,
        receivedDate,
        recipient,
        mainImage,
        image1,
        image2,
        image3,
        detailUrl,
        String(existing[10] || '').trim() || today,
        posterKey
      ];
    }

    if (mode === 'courseadmin') {
      const courseCategory = String(data.courseCategory || '').trim();
      const courseName = String(data.courseName || '').trim();
      if (!courseCategory) {
        throw new Error('กรุณาเลือกหลักสูตร');
      }
      if (!courseName) {
        throw new Error('กรุณากรอกชื่อหลักสูตร');
      }
      const courseOptions = getContentAdminOptions_('courseadmin');
      if (courseOptions.length && courseOptions.indexOf(courseCategory) === -1) {
        throw new Error('หลักสูตรไม่อยู่ในรายการ setting!AJ2:AJ');
      }

      let fileUrl = String(data.fileUrl || '').trim();
      if (String(data.fileData || '').trim()) {
        fileUrl = uploadContentAdminFile_(data.fileData, data.fileName, 'course-file', false);
      }
      if (!fileUrl) throw new Error('กรุณาระบุไฟล์');

      values = [
        order,
        courseName,
        posterKey,
        String(existing[3] || '').trim() || today,
        fileUrl,
        posterKey,
        courseCategory
      ];
    }

    if (!values) throw new Error('Invalid admin mode: ' + mode);

    if (config.shared && !editing) {
      // แต่ละเว็บไซต์เป็นคนละ Apps Script deployment จึงใช้ appendRow
      // เพื่อลดความเสี่ยงเขียนทับกันเมื่อเพิ่มข้อมูลพร้อมกัน
      sheet.appendRow(values);
      rowNumber = sheet.getLastRow();
    } else {
      sheet.getRange(rowNumber, 1, 1, config.columns).setValues([values]);
    }

    SpreadsheetApp.flush();
    clearContentAdminCaches_(mode);

    return {
      rowNumber: rowNumber,
      posterKey: posterKey,
      item: listContentAdminItems_(mode).filter(function(item) {
        return Number(item.rowNumber) === Number(rowNumber);
      })[0] || null
    };
  } finally {
    lock.releaseLock();
  }
}

function deleteContentAdminItem_(mode, data) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);

  try {
    const sheet = getContentAdminSheet_(mode);
    const config = getContentAdminConfig_(mode);
    const rowNumber = Number(data.rowNumber || 0);

    if (rowNumber < 2 || rowNumber > sheet.getLastRow()) {
      throw new Error('ไม่พบแถวข้อมูลที่ต้องการลบ');
    }

    if (config.shared) {
      const posterKey = getContentPosterKey_();
      const existingKey = String(
        sheet.getRange(rowNumber, config.keyColumn).getDisplayValue() || ''
      ).trim();

      if (existingKey !== posterKey) {
        throw new Error('ไม่อนุญาตให้ลบข้อมูลของคีย์อื่น');
      }
    }

    sheet.deleteRow(rowNumber);
    renumberContentAdminSheet_(sheet);
    SpreadsheetApp.flush();
    clearContentAdminCaches_(mode);

    return { deleted: true };
  } finally {
    lock.releaseLock();
  }
}

function renumberContentAdminSheet_(sheet) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return;

  const values = [];
  for (let row = 2; row <= lastRow; row++) {
    values.push([row - 1]);
  }
  sheet.getRange(2, 1, values.length, 1).setValues(values);
}

}

function uploadContentAdminFile_(dataUrl, filename, prefix, imageOutput) {
  const value = String(dataUrl || '').trim();
  if (!value) return '';

  const match = value.match(/^data:([^;]+);base64,(.+)$/);
  if (!match) throw new Error('ข้อมูลไฟล์อัปโหลดไม่ถูกต้อง');

  const contentType = String(match[1] || 'application/octet-stream').trim();
  const bytes = Utilities.base64Decode(match[2]);

  if (bytes.length > 15 * 1024 * 1024) {
    throw new Error('ไฟล์มีขนาดใหญ่เกิน 15 MB');
  }

  const folderId = getWebsiteImageFolderId_();
  const safeBase = String(filename || (prefix || 'upload'))
    .replace(/[\\\\/:*?"<>|]/g, '_')
    .replace(/^\.+/, '') || 'upload';
  const safeName = String(prefix || 'content') + '-' + Date.now() + '-' + safeBase;
  const blob = Utilities.newBlob(bytes, contentType, safeName);

  let file;
  try {
    file = Drive.Files.create(
      {
        name: safeName,
        parents: [folderId]
      },
      blob,
      {
        supportsAllDrives: true,
        fields: 'id,name'
      }
    );
  } catch (err) {
    throw new Error(
      'อัปโหลดไฟล์เข้า Google Drive ไม่สำเร็จ: ' +
      (err && err.message ? err.message : err)
    );
  }

  try {
    Drive.Permissions.create(
      { role: 'reader', type: 'anyone' },
      file.id,
      { supportsAllDrives: true }
    );
  } catch (err) {
    try { Drive.Files.remove(file.id); } catch (ignore) {}
    throw new Error(
      'ตั้งสิทธิ์ไฟล์เป็นสาธารณะไม่สำเร็จ: ' +
      (err && err.message ? err.message : err)
    );
  }

  if (imageOutput || contentType.indexOf('image/') === 0) {
    return 'https://lh3.googleusercontent.com/d/' + file.id;
  }

  return 'https://drive.google.com/file/d/' + file.id + '/view?usp=sharing';
}


// ============================================================
// Admin storage meter for index.html
// Calculates all non-trashed files in the configured folder and subfolders.
// Cached so entering Admin mode does not slow down the page.
// ============================================================
function getAdminStorageUsage_(forceRefresh) {
  const cache = CacheService.getScriptCache();
  const cacheKey = 'ADMIN_STORAGE_USAGE_V4_D11_' + ADMIN_STORAGE_METER_FOLDER_ID + '_' + ADMIN_STORAGE_LIMIT_GB;

  if (!forceRefresh) {
    const cached = cache.get(cacheKey);
    if (cached) {
      try { return JSON.parse(cached); } catch (ignore) {}
    }
  }

  if (!ADMIN_STORAGE_METER_FOLDER_ID) throw new Error('ไม่พบ Folder ID สำหรับ adminStorageMeter ที่ DATABSE_MAIN_ID > Sheet!D11');
  const root = DriveApp.getFolderById(ADMIN_STORAGE_METER_FOLDER_ID);
  const folders = [root];
  let usedBytes = 0;
  let fileCount = 0;
  let folderCount = 1;

  while (folders.length) {
    const folder = folders.pop();
    const files = folder.getFiles();

    while (files.hasNext()) {
      const file = files.next();
      try {
        if (typeof file.isTrashed === 'function' && file.isTrashed()) continue;
        usedBytes += Math.max(0, Number(file.getSize()) || 0);
        fileCount++;
      } catch (ignore) {}
    }

    const children = folder.getFolders();
    while (children.hasNext()) {
      folders.push(children.next());
      folderCount++;
    }
  }

  const percent = Math.min(100, Math.max(0, (usedBytes / ADMIN_STORAGE_LIMIT_BYTES) * 100));
  const data = {
    folderId: ADMIN_STORAGE_METER_FOLDER_ID,
    usedBytes: usedBytes,
    limitBytes: ADMIN_STORAGE_LIMIT_BYTES,
    limitLabel: ADMIN_STORAGE_LIMIT_GB + ' GB',
    percent: percent,
    fileCount: fileCount,
    folderCount: folderCount,
    updatedAt: new Date().toISOString()
  };

  try { cache.put(cacheKey, JSON.stringify(data), 300); } catch (ignore) {}
  return data;
}

function adminLogin_(username, password) {
  username = String(username || '').trim();
  password = String(password || '');
  if (!username || !password) throw new Error('กรุณากรอก Username และ Password');

  const attemptCache = CacheService.getScriptCache();
  const attemptKey = 'admin-attempt:' + Utilities.base64EncodeWebSafe(username.toLowerCase()).slice(0, 80);
  const attempts = Number(attemptCache.get(attemptKey) || 0);
  if (attempts >= 8) throw new Error('เข้าสู่ระบบผิดหลายครั้ง กรุณารอ 10 นาทีแล้วลองใหม่');

  const sheet = getAdminLoginSheet_();
  const lastRow = sheet.getLastRow();
  const values = lastRow >= 2 ? sheet.getRange(2, 1, lastRow - 1, 3).getDisplayValues() : [];
  const row = values.find(function (item) {
    return String(item[1] || '').trim() === username && String(item[2] || '') === password;
  });

  if (!row) {
    attemptCache.put(attemptKey, String(attempts + 1), 600);
    throw new Error('Username หรือ Password ไม่ถูกต้อง');
  }

  attemptCache.remove(attemptKey);
  const token = Utilities.getUuid() + Utilities.getUuid().replace(/-/g, '');
  CacheService.getScriptCache().put('admin-session:' + token, JSON.stringify({ username: row[1], email: row[0] }), 21600);
  return { success: true, token: token, username: String(row[1] || ''), expiresIn: 21600 };
}

function requireAdminSession_(token) {
  token = String(token || '').trim();
  if (!token || !CacheService.getScriptCache().get('admin-session:' + token)) {
    throw new Error('Session หมดอายุ กรุณาเข้าสู่ระบบใหม่');
  }
  return true;
}

function sendAdminCredentials_(email) {
  email = String(email || '').trim().toLowerCase();
  if (!email || email.indexOf('@') === -1) throw new Error('กรุณากรอก Email ให้ถูกต้อง');

  const cache = CacheService.getScriptCache();
  const cooldownKey = 'admin-forgot:' + Utilities.base64EncodeWebSafe(email).slice(0, 80);
  if (cache.get(cooldownKey)) throw new Error('เพิ่งส่ง Email ไปแล้ว กรุณารอ 5 นาที');

  const sheet = getAdminLoginSheet_();
  const lastRow = sheet.getLastRow();
  const values = lastRow >= 2 ? sheet.getRange(2, 1, lastRow - 1, 3).getDisplayValues() : [];
  const row = values.find(function (item) { return String(item[0] || '').trim().toLowerCase() === email; });
  if (!row) throw new Error('ไม่พบ Email นี้ในระบบ');

  MailApp.sendEmail({
    to: String(row[0] || '').trim(),
    subject: 'ข้อมูลเข้าสู่ระบบจัดการเว็บไซต์',
    name: 'ระบบจัดการเว็บไซต์',
    htmlBody: '<div style="font-family:Arial,sans-serif;line-height:1.7"><h2>ข้อมูลเข้าสู่ระบบจัดการเว็บไซต์</h2><p><b>Username:</b> ' + escapeHtmlEmail_(row[1]) + '</p><p><b>Password:</b> ' + escapeHtmlEmail_(row[2]) + '</p><p>หากคุณไม่ได้เป็นผู้ร้องขอ กรุณาแจ้งผู้ดูแลระบบ</p></div>'
  });
  cache.put(cooldownKey, '1', 300);
  return { success: true };
}

function escapeHtmlEmail_(value) {
  return String(value || '').replace(/[&<>"']/g, function (character) {
    return ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' })[character];
  });
}

/**
 * รายการ Dropdown สำหรับ main-nav จากชีต setting กลาง
 * H1/J1 = ชื่อปุ่ม, H2:I = สกร.ระดับตำบล, J2:K = ห้องสมุด
 */
function getMainNavMenus_() {
  // ใช้เมนูกลางจาก Spreadsheet อำเภอสำหรับทั้ง 3 เว็บไซต์
  const ss = SpreadsheetApp.openById(DATABSE_MAIN_ID);
  const sheet = ss.getSheetByName(NAV_MENU_SHEET_NAME);

  if (!sheet) {
    throw new Error('ไม่พบชีต ' + NAV_MENU_SHEET_NAME + ' สำหรับเมนูหลัก');
  }

  // H1 = ชื่อปุ่ม สกร.ระดับตำบล, J1 = ชื่อปุ่ม ห้องสมุด
  const headers = sheet.getRange(1, 8, 1, 4).getDisplayValues()[0];
  const districtTitle = String(headers[0] || '').trim() || 'สกร.ระดับตำบล';
  const libraryTitle = String(headers[2] || '').trim() || 'ห้องสมุด';
  const lastRow = sheet.getLastRow();

  if (lastRow < 2) {
    return {
      districtTitle: districtTitle,
      libraryTitle: libraryTitle,
      districts: [],
      libraries: []
    };
  }

  // H2:H = ชื่อตัวเลือก, I2:I = URL
  // J2:J = ชื่อตัวเลือก, K2:K = URL
  const values = sheet
    .getRange(2, 8, lastRow - 1, 4)
    .getDisplayValues();

  const districts = [];
  const libraries = [];

  values.forEach(function (row) {
    const districtName = String(row[0] || '').trim();
    const districtUrl = String(row[1] || '').trim();
    const libraryName = String(row[2] || '').trim();
    const libraryUrl = String(row[3] || '').trim();

    if (districtName && districtUrl) {
      districts.push({ name: districtName, url: districtUrl });
    }

    if (libraryName && libraryUrl) {
      libraries.push({ name: libraryName, url: libraryUrl });
    }
  });

  return {
    districtTitle: districtTitle,
    libraryTitle: libraryTitle,
    districts: districts,
    libraries: libraries
  };
}

 * B = ชื่อหลักสูตร, C = ผู้โพสต์, D = วันที่อัปโหลดไฟล์, E = URL ดาวน์โหลดไฟล์,
 * F = คีย์ผู้โพสต์, G = หลักสูตร (ตัวเลือกจาก setting!AJ2:AJ)
 */
function getCourseItems_() {
  const posterKey = getContentPosterKey_();
  const sheet = getContentAdminSheet_('courseadmin');
  if (!sheet || sheet.getLastRow() < 2) return [];

  const values = sheet.getRange(2, 1, sheet.getLastRow() - 1, 7).getDisplayValues();
  const items = [];

  values.forEach(function(row) {
    const courseName = String(row[1] || '').trim();
    const teacherName = String(row[2] || '').trim();
    const uploadDate = String(row[3] || '').trim();
    const fileUrl = String(row[4] || '').trim();
    const key = String(row[5] || '').trim();
    const courseCategory = String(row[6] || '').trim();
    if (posterKey && key !== posterKey) return;
    if (!courseName && !teacherName && !uploadDate && !fileUrl) return;

    items.push({
      courseCategory: courseCategory,
      courseName: courseName,
      teacherName: teacherName,
      uploadDate: uploadDate,
      fileUrl: fileUrl,
      area: key,
      posterKey: key,
      spreadsheetId: DATABSE_MAIN_ID
    });
  });

  return items;
}


/**
 * อ่านโพสต์ Facebook จากฐานข้อมูลกลาง
 * DATABSE_MAIN_ID > facebook!A2:C
 * A = คีย์ผู้โพสต์, B = iframe embed Facebook, C = วันที่
 */
function getFacebookItems_() {
  // ใช้ฐานข้อมูล Facebook กลางเพียงแห่งเดียว: DATABSE_MAIN_ID > facebook!A:C
  // A = คีย์ผู้โพสต์, B = iframe embed Facebook, C = วันที่
  const posterKey = getFacebookPosterKey_();
  const ss = SpreadsheetApp.openById(DATABSE_MAIN_ID);
  const sheet = ss.getSheetByName('facebook');
  if (!sheet || sheet.getLastRow() < 2) return [];

  const values = sheet
    .getRange(2, 1, sheet.getLastRow() - 1, 3)
    .getDisplayValues();

  const items = [];

  values.forEach(function(value) {
    const area = String(value[0] || '').trim();
    const iframeCode = String(value[1] || '').trim();
    const date = String(value[2] || '').trim();

    if (area !== posterKey) return;
    if (!iframeCode) return;

    const srcMatch = iframeCode.match(/<iframe[^>]*\bsrc\s*=\s*["']([^"']+)["']/i);
    if (!srcMatch) return;

    const embedUrl = String(srcMatch[1] || '')
      .replace(/&(?:amp|#38|#x26);/gi, '&')
      .trim();

    if (!/^https:\/\/(?:[a-z0-9-]+\.)*facebook\.com\/plugins\/post\.php\?/i.test(embedUrl)) {
      return;
    }

    const hrefMatch = embedUrl.match(/[?&]href=([^&]+)/i);
    let facebookUrl = '';

    if (hrefMatch) {
      try {
        facebookUrl = decodeURIComponent(hrefMatch[1]);
      } catch (ignore) {
        facebookUrl = hrefMatch[1];
      }
    }

    if (!/^https:\/\/(?:[a-z0-9-]+\.)*facebook\.com\//i.test(facebookUrl)) {
      return;
    }

    items.push({
      area: area,
      date: date,
      embedUrl: embedUrl,
      facebookUrl: facebookUrl,
      spreadsheetId: DATABSE_MAIN_ID
    });
  });

  return items;
}

function extractSpreadsheetId_(value) {
  const text = String(value || '').trim();
  if (!text) return '';

  const urlMatch = text.match(/\/spreadsheets\/d\/([a-zA-Z0-9_-]+)/);
  if (urlMatch) return urlMatch[1];

  return /^[a-zA-Z0-9_-]{20,}$/.test(text) ? text : '';
}

/**
 * ใช้แทนฟังก์ชัน getNewsSlides() เดิมใน Google Apps Script
 * ชีต: news_page
 * B = หัวข้อหลัก
 * C = URL รูปภาพโปสเตอร์ข่าว
 * D = รายละเอียดโดยย่อ
 * E = URL รายละเอียด (เว้นว่างได้)
 * F = วันที่ลงข่าว
 * J1 = block เพื่อสไลด์หลายภาพ / none เพื่อแสดงภาพเดียว
 */
function getNewsSlides() {
  const ss = SpreadsheetApp.openById(DATABSE_MAIN_TEATHER_ID);
  const sheet = ss.getSheetByName('news_page');

  if (!sheet) {
    throw new Error('ไม่พบชีต news_page');
  }

  const mode = String(sheet.getRange('J1').getDisplayValue() || 'none')
    .trim()
    .toLowerCase();

  const lastRow = sheet.getLastRow();

  if (lastRow < 2) {
    return {
      mode: mode,
      slides: []
    };
  }

  // อ่าน B2:F ตามจำนวนแถวจริง
  const rows = sheet
    .getRange(2, 2, lastRow - 1, 5)
    .getDisplayValues();

  let slides = rows
    .map(function (row, index) {
      return {
        newsNo: index + 1,
        title: String(row[0] || '').trim(),
        image: String(row[1] || '').trim(),
        detail: String(row[2] || '').trim(),
        detailUrl: String(row[3] || '').trim(),
        date: String(row[4] || '').trim()
      };
    })
    .filter(function (item) {
      return item.image !== '';
    });

  // none = แสดงเฉพาะข่าวแรก
  if (mode !== 'block' && slides.length > 1) {
    slides = slides.slice(0, 1);
  }

  return {
    mode: mode,
    slides: slides
  };
}

function runNewsAdminAction_(action, data) {
  action = String(action || '').trim().toLowerCase();
  if (action === 'list') return getNewsAdminData_();

  );

function saveNewsAdmin_(data) {
  data = data || {};
  const title = String(data.title || '').trim();
  let imageUrl = String(data.imageUrl || '').trim();
  const imageData = String(data.imageData || '').trim();
  const detail = String(data.detail || '').trim();
  const detailUrl = String(data.detailUrl || '').trim();

  // ช่อง newsImage รองรับทั้ง URL เดิมและการอัปโหลดรูปจากเครื่อง
  if (imageData) {
    if (!/^data:image\/[a-zA-Z0-9.+-]+;base64,/i.test(imageData)) {
      throw new Error('ไฟล์อัปโหลดต้องเป็นรูปภาพเท่านั้น');
    }
    imageUrl = uploadNewsAdminImage_(imageData, String(data.imageName || 'news-poster.jpg'));
  }

  if (!title || !imageUrl || !detail) {
    throw new Error('กรุณากรอกหัวข้อ รายละเอียด และระบุ URL รูปภาพหรืออัปโหลดรูป');
  }
  // newsUrl / detailUrl เว้นว่างได้ แต่ถ้าระบุต้องเป็น URL http/https
  if (detailUrl && !/^https?:\/\//i.test(detailUrl)) {
    throw new Error('URL กดดูรายละเอียดไม่ถูกต้อง');
  }
  const sheet = getNewsAdminSheet_();
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const rowNumber = Number(data.rowNumber || 0);
    if (rowNumber) {
      if (rowNumber < 2 || rowNumber > sheet.getLastRow()) throw new Error('ไม่พบแถวข่าวนี้');
      sheet.getRange(rowNumber, 2, 1, 4).setValues([[title,imageUrl,detail,detailUrl]]);
    } else {
      sheet.insertRowAfter(1);
      sheet.getRange(2, 1, 1, 6).setValues([[1,title,imageUrl,detail,detailUrl,formatNewsThaiDate_(new Date())]]);
      reorderNewsAdmin_(sheet);
    }
    SpreadsheetApp.flush();
  } finally { lock.releaseLock(); }
  clearHomeFastCache_();
  return true;
}

function deleteNewsAdmin_(rowNumber) {
  const sheet = getNewsAdminSheet_();
  const row = Number(rowNumber);
  if (row < 2 || row > sheet.getLastRow()) throw new Error('ไม่พบแถวข่าวนี้');
  const lock = LockService.getScriptLock(); lock.waitLock(10000);
  try { sheet.deleteRow(row); reorderNewsAdmin_(sheet); SpreadsheetApp.flush(); } finally { lock.releaseLock(); }
  clearHomeFastCache_();
  return true;
}

function moveNewsAdmin_(rowNumber, direction) {
  const sheet = getNewsAdminSheet_();
  const row = Number(rowNumber), lastRow = sheet.getLastRow();
  if (row < 2 || row > lastRow) throw new Error('ไม่พบแถวข่าวนี้');
  direction = String(direction || '').toLowerCase();
  const target = direction === 'up' ? row - 1 : direction === 'down' ? row + 1 : row;
  if (target < 2 || target > lastRow || target === row) return true;
  const lock = LockService.getScriptLock(); lock.waitLock(10000);
  try {
    const current = sheet.getRange(row,1,1,6).getValues();
    const other = sheet.getRange(target,1,1,6).getValues();
    sheet.getRange(row,1,1,6).setValues(other); sheet.getRange(target,1,1,6).setValues(current);
    reorderNewsAdmin_(sheet); SpreadsheetApp.flush();
  } finally { lock.releaseLock(); }
  clearHomeFastCache_();
  return true;
}

function setNewsAdminMode_(value) {
  const mode = String(value || '').toLowerCase() === 'block' ? 'block' : 'none';
  getNewsAdminSheet_().getRange('J1').setValue(mode);
  clearHomeFastCache_();
  return mode;
}

function reorderNewsAdmin_(sheet) {
  const count = sheet.getLastRow() - 1;
  if (count < 1) return;
  const values = Array.from({length:count}, function (_, index) { return [index + 1]; });
  sheet.getRange(2,1,count,1).setValues(values);
}

function formatNewsThaiDate_(date) {
  const months = ['มกราคม','กุมภาพันธ์','มีนาคม','เมษายน','พฤษภาคม','มิถุนายน','กรกฎาคม','สิงหาคม','กันยายน','ตุลาคม','พฤศจิกายน','ธันวาคม'];
  return date.getDate() + ' ' + months[date.getMonth()] + ' ' + (date.getFullYear() + 543);
}

function normalizePublicImageUrl_(url) {
  const value = String(url || '').trim();
  if (!value) return '';

  const driveMatch = value.match(/[-\w]{25,}/);
  if (value.includes('drive.google.com') && driveMatch) {
    return 'https://lh3.googleusercontent.com/d/' + driveMatch[0];
  }

  return value;
}

function uploadStudentPhoto_(base64, filename) {
  if (!base64) return "";

  const match = String(base64).match(/^data:(image\/[a-zA-Z0-9.+-]+);base64,(.+)$/);
  if (!match) {
    throw new Error("รูปภาพมีรูปแบบไม่ถูกต้อง");
  }

  const contentType = match[1];
  const bytes = Utilities.base64Decode(match[2]);
  if (bytes.length > 5 * 1024 * 1024) {
    throw new Error("รูปภาพต้องมีขนาดไม่เกิน 5 MB");
  }

  const safeName = String(filename || ("student-" + Date.now() + ".jpg"))
    .replace(/[\\/:*?"<>|]/g, "_");
  const blob = Utilities.newBlob(bytes, contentType, safeName);

  const file = Drive.Files.create(
    {
      name: safeName,
      parents: [DB_ID]
    },
    blob,
    {
      supportsAllDrives: true,
      fields: 'id,name'
    }
  );

  Drive.Permissions.create(
    {
      role: "reader",
      type: "anyone"
    },
    file.id,
    { supportsAllDrives: true }
  );

  return "https://lh3.googleusercontent.com/d/" + file.id;
}

function createSheet_(ss, name, headers) {

function ensureHeader_(sheet, headerName) {
  if (!sheet) return;
  const lastColumn = Math.max(sheet.getLastColumn(), 1);
  const headers = sheet.getRange(1, 1, 1, lastColumn).getDisplayValues()[0];
  if (!headers.includes(headerName)) {
    sheet.getRange(1, lastColumn + 1).setValue(headerName);
  }
}

function uid_(prefix) {

function getData_(sheetName) {

    const obj = {};
    headers.forEach((h, i) => obj[h] = r[i]);
    return obj;
  });
}

function registerStudent(data) {
  setupLearningBaseSheets_();

  const fullname = String(data.fullname || "").trim();
  const phone = String(data.phone || "").trim();
  const address = String(data.address || "").trim();
  const photoBase64 = String(data.photoBase64 || "");
  const photoName = String(data.photoName || "").trim();

  if (!fullname) {
    return { ok: false, message: "กรุณากรอกชื่อ-นามสกุล" };
  }

  if (!phone) {
    return { ok: false, message: "กรุณากรอกเบอร์โทร" };
  }

  if (!/^[0-9]{9,10}$/.test(phone)) {
    return { ok: false, message: "เบอร์โทรต้องเป็นตัวเลข 9-10 หลัก" };
  }

  const students = getData_(SHEETS.students);
  const exists = students.find(s =>
    normalizePhone_(s.phone) === normalizePhone_(phone)
  );

  if (exists) {
    return {
      ok: false,
      message: "เบอร์โทรนี้ถูกลงทะเบียนแล้ว"
    };
  }

  let photo = "";
  if (photoBase64) {
    photo = uploadStudentPhoto_(photoBase64, photoName);
  }

  const studentId = uid_("STU");

  append_(SHEETS.students, [
    studentId,
    fullname,
    phone,
    address,
    new Date(),
    photo
  ]);

  // Google Sheets may auto-convert a phone such as 0619042117 to a number
  // and drop the first 0. Force the newly saved phone cell to plain text.
  const studentSheet = ss_().getSheetByName(SHEETS.students);
  const newStudentRow = studentSheet.getLastRow();
  studentSheet
    .getRange(newStudentRow, 3)
    .setNumberFormat('@')
    .setValue(phone);

  return {
    ok: true,
    student: {
      studentId,
      fullname,
      phone,
      address,
      photo
    },
    message: "ลงทะเบียนสำเร็จ"
  };
}

  if (!/^0\d{9}$/.test(phone)) {
    return {
      ok: false,
      message: 'เบอร์โทรต้องขึ้นต้นด้วย 0 และมี 10 หลัก'
    };
  }

  const sheet = ss_().getSheetByName(
    SHEETS.students
  );

  if (!sheet) {
    return {
      ok: false,
      message: 'ไม่พบชีต students'
    };
  }

  const values = sheet
    .getDataRange()
    .getValues();

  if (values.length < 2) {
    return {
      ok: false,
      message: 'ไม่พบข้อมูลผู้ใช้งาน'
    };
  }

  const headers = values[0].map(value =>
    String(value || '').trim()
  );

  const studentIdCol =
    headers.indexOf('studentId');

  const fullnameCol =
    headers.indexOf('fullname');

  const phoneCol =
    headers.indexOf('phone');

  const addressCol =
    headers.indexOf('address');

  const photoCol =
    headers.indexOf('photo');

  if (
    studentIdCol === -1 ||
    fullnameCol === -1 ||
    phoneCol === -1 ||
    addressCol === -1 ||
    photoCol === -1
  ) {
    return {
      ok: false,
      message: 'โครงสร้างชีต students ไม่ถูกต้อง'
    };
  }

  let targetRow = -1;

  for (let i = 1; i < values.length; i++) {
    if (
      String(
        values[i][studentIdCol] || ''
      ).trim() === studentId
    ) {
      targetRow = i + 1;
      break;
    }
  }

  if (targetRow === -1) {
    return {
      ok: false,
      message: 'ไม่พบข้อมูลผู้ใช้งาน'
    };
  }

  /*
   * ป้องกันไม่ให้เบอร์โทรซ้ำกับบัญชีอื่น
   */
  for (let i = 1; i < values.length; i++) {
    const rowStudentId = String(
      values[i][studentIdCol] || ''
    ).trim();

    const rowPhone = normalizePhone_(
      values[i][phoneCol]
    );

    if (
      rowStudentId !== studentId &&
      rowPhone === phone
    ) {
      return {
        ok: false,
        message: 'เบอร์โทรนี้ถูกใช้งานโดยบัญชีอื่นแล้ว'
      };
    }
  }

  let photo = String(
    sheet
      .getRange(targetRow, photoCol + 1)
      .getDisplayValue() || ''
  ).trim();

  /*
   * ถ้ากดลบรูป ให้ล้าง URL เดิม
   */
  if (removePhoto) {
    photo = '';
  }

  /*
   * ถ้าเลือกรูปใหม่ ให้อัปโหลดและแทน URL เดิม
   */
  if (photoBase64) {
    photo = uploadStudentPhoto_(
      photoBase64,
      photoName ||
        `student-${studentId}-${Date.now()}.jpg`
    );
  }

  sheet
    .getRange(targetRow, fullnameCol + 1)
    .setValue(fullname);

  sheet
    .getRange(targetRow, phoneCol + 1)
    .setNumberFormat('@')
    .setValue(phone);

  sheet
    .getRange(targetRow, addressCol + 1)
    .setValue(address);

  sheet
    .getRange(targetRow, photoCol + 1)
    .setValue(photo);

  SpreadsheetApp.flush();

  return {
    ok: true,
    message: 'บันทึกข้อมูลโปรไฟล์เรียบร้อยแล้ว',
    student: {
      studentId,
      fullname,
      phone,
      address,
      photo
    }
  };
}

function findTeacherAdminAccount_(username, password, checkPassword) {
  username = String(username || "").trim();
  password = String(password || "");
  if (!username) return null;

  // ใช้แหล่งบัญชีเดียวกับ Login ผู้ดูแลระบบของเว็บไซต์นั้นโดยตรง
  // รูปแบบบัญชี: A = Email/Name, B = Username, C = Password
  const sheet = getAdminLoginSheet_();
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return null;

  const values = sheet.getRange(2, 1, lastRow - 1, 3).getDisplayValues();
  for (let i = 0; i < values.length; i++) {
    const account = normalizeTeacherAdminAccount_(values[i]);
    if (account.username !== username) continue;
    if (checkPassword && account.password !== password) continue;
    return account;
  }
  return null;
}

function makeTeacherIdFromAdminUsername_(username) {
  const value = String(username || "").trim().toLowerCase();
  const bytes = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256,
    value,
    Utilities.Charset.UTF_8
  );
  const hex = bytes.map(function(byte) {
    const n = byte < 0 ? byte + 256 : byte;
    return ("0" + n.toString(16)).slice(-2);
  }).join("").toUpperCase();
  return "TCH_AUTH_" + hex.slice(0, 16);
}

function getOrCreateTeacherProfileFromAdminAccount_(account) {
  if (!account || !account.username) {
    throw new Error("ไม่พบบัญชีผู้ดูแลระบบ");
  }

  setupLearningBaseSheets_();

  const lock = LockService.getScriptLock();
  lock.waitLock(5000);
  try {
    const teachers = getData_(SHEETS.teachers);
    let teacher = teachers.find(function(item) {
      return String(item.username || "").trim() === account.username;
    });

    // ถ้ามีข้อมูลครูเดิม ให้ใช้ teacherId / ชื่อฐาน / รูป / Email เดิมต่อ
    // เพื่อให้กิจกรรมเก่าและประวัติเดิมยังผูกกับเจ้าของคนเดิมครบถ้วน
    if (teacher) {
      const merged = {
        teacherId: String(teacher.teacherId || "").trim() || makeTeacherIdFromAdminUsername_(account.username),
        username: account.username,
        teacherName: String(teacher.teacherName || account.nameOrEmail || account.username).trim(),
        teacherImage: String(teacher.teacherImage || "").trim(),
        baseName: String(teacher.baseName || "").trim(),
        email: String(teacher.email || (/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(account.nameOrEmail) ? account.nameOrEmail : "")).trim(),
        lineUserId: String(teacher.lineUserId || "").trim(),
        createdAt: teacher.createdAt || ""
      };
      return sanitizeTeacherProfile_(merged);
    }

    // บัญชีใหม่ไม่ต้องเพิ่มคอลัมน์ในชีต Login ใหม่
    // ระบบสร้างเฉพาะ Profile ภายในชีต teachers ของ DB_ID เพื่อเก็บ teacherId
    // และข้อมูลประกอบที่ระบบกิจกรรมจำเป็นต้องใช้ โดยไม่เก็บ Password ซ้ำ
    const teacherId = makeTeacherIdFromAdminUsername_(account.username);
    const teacherName = String(account.nameOrEmail || account.username).trim();
    const email = /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(account.nameOrEmail)
      ? account.nameOrEmail
      : "";

    append_(SHEETS.teachers, [
      teacherId,
      account.username,
      "", // ไม่เก็บ Password ซ้ำใน teachers
      teacherName,
      "", // teacherImage: ใช้ค่าว่างได้
      "", // baseName: ใช้ค่าว่างได้ ระบบส่วนอื่นยังทำงานต่อได้
      email,
      "", // lineUserId
      new Date()
    ]);

    return sanitizeTeacherProfile_({
      teacherId: teacherId,
      username: account.username,
      teacherName: teacherName,
      teacherImage: "",
      baseName: "",
      email: email,
      lineUserId: "",
      createdAt: new Date()
    });
  } finally {
    lock.releaseLock();
  }
}

function teacherLogin(username, password) {
  username = String(username || "").trim();
  password = String(password || "");

  if (!username || !password) {
    return { ok: false, message: "กรุณากรอก Username และ Password" };
  }

  // จำกัดการลองรหัสผ่านผิดเหมือนระบบผู้ดูแล
  const cache = CacheService.getScriptCache();
  const attemptKey = "teacher-admin-attempt:" +
    Utilities.base64EncodeWebSafe(username.toLowerCase()).slice(0, 80);
  const attempts = Number(cache.get(attemptKey) || 0);
  if (attempts >= 8) {
    return { ok: false, message: "เข้าสู่ระบบผิดหลายครั้ง กรุณารอ 10 นาทีแล้วลองใหม่" };
  }

  const account = findTeacherAdminAccount_(username, password, true);
  if (!account) {
    cache.put(attemptKey, String(attempts + 1), 600);
    return { ok: false, message: "Username หรือ Password ไม่ถูกต้อง" };
  }

  cache.remove(attemptKey);
  const teacher = getOrCreateTeacherProfileFromAdminAccount_(account);

  return {
    ok: true,
    teacher: teacher,
    message: "เข้าสู่ระบบสำเร็จด้วยบัญชีผู้ดูแลระบบ"
  };
}

function getTeacherDashboard(teacherId) {
  const tid = String(teacherId || "").trim();

  const activities = getData_(SHEETS.activities)
    .filter(a => String(a.teacherId || "").trim() === tid)
    .reverse();

  const joins = getData_(SHEETS.joins);
  const students = getData_(SHEETS.students);
  const scores = getData_(SHEETS.score);

  return activities.map(a => {
    const actId = String(a.activityId || "").trim();

    const applicants = joins
      .filter(j => {
        const sid = String(j.studentId || "").trim();
        const aid = String(j.activityId || "").trim();

        const alreadySaved = scores.some(s =>
          String(s.studentId || "").trim() === sid &&
          String(s.activityId || "").trim() === aid
        );

        return aid === actId && !alreadySaved;
      })
      .map(j => {
        const sid = String(j.studentId || "").trim();
        const student = students.find(s =>
          String(s.studentId || "").trim() === sid
        );

        return {
          ...j,
          student: student || {}
        };
      });

    return { ...a, applicants };
  });
}

function normalizePhone_(phone) {
  let tel = String(phone || "").trim();
  tel = tel.replace(/\D/g, "");

  if (!tel) return "";

  if (tel.length === 9) tel = "0" + tel;

  return tel;
}

function saveActualHours(data) {

  const teacherId = String(data.teacherId || "").trim();

function normalizeLearningHeader_(value) {
  return String(value || '')
    .trim()
    .toLowerCase()
    .replace(/[\s\-_\/()]+/g, '');
}

function findLearningSheet_(ss, names) {
  const wanted = names.map(normalizeLearningHeader_);
  return ss.getSheets().find(function(sheet) {
    return wanted.indexOf(normalizeLearningHeader_(sheet.getName())) !== -1;
  }) || null;
}

function sheetRowsAsObjects_(sheet) {
  if (!sheet || sheet.getLastRow() < 2 || sheet.getLastColumn() < 1) return [];
  const values = sheet.getDataRange().getDisplayValues();
  const headers = values[0].map(function(value) { return String(value || '').trim(); });
  return values.slice(1)
    .filter(function(row) { return row.some(function(value) { return String(value || '').trim() !== ''; }); })
    .map(function(row) {
      const item = {};
      headers.forEach(function(header, index) {
        item[header] = row[index];
        item[normalizeLearningHeader_(header)] = row[index];
      });
      return item;
    });
}

function pickLearningValue_(row, aliases, fallback) {
  for (let i = 0; i < aliases.length; i++) {
    const key = normalizeLearningHeader_(aliases[i]);
    if (Object.prototype.hasOwnProperty.call(row, key)) {
      const value = row[key];
      if (String(value || '').trim() !== '') return value;
    }
  }
  return fallback === undefined ? '' : fallback;
}

function getLearningSettings_(ss) {
  const sheet = findLearningSheet_(ss, ['settings','setting','ตั้งค่า','การตั้งค่า']);

  const values = sheet.getDataRange().getDisplayValues();

  // รูปแบบ key/value ในคอลัมน์ A:B
  values.forEach(function(row, index) {
    const key = normalizeLearningHeader_(row[0]);
    const val = String(row[1] || '').trim();
    if (key && val && index > 0) settings[key] = val;
  });

  // รูปแบบหัวตารางแถว 1 และข้อมูลแถว 2
  if (values.length >= 2) {
    values[0].forEach(function(header, index) {
      const key = normalizeLearningHeader_(header);
      const val = String(values[1][index] || '').trim();
      if (key && val) settings[key] = val;
    });
  }

  return {
    system_name: settings.systemname || settings['ชื่อระบบ'] || settings.title || 'แหล่งเรียนรู้ในพื้นที่',
    organization: settings.organization || settings['หน่วยงาน'] || '',

function getReportData_(settingSheetName, urlCell, teacherCol, districtCol, reportName, mainSS) {
  try {
    mainSS = mainSS || SpreadsheetApp.openById(DATABSE_MAIN_TEATHER_ID);
    const settingSh = mainSS.getSheetByName(settingSheetName);

    if (!settingSh) {
      return {
        ok: false,
        message: reportName + " : ไม่พบชีตตั้งค่า " + settingSheetName,
        levels: {}
      };
    }

    const url = String(settingSh.getRange(urlCell).getDisplayValue() || "").trim();

    if (!url) {
      return {
        ok: false,
        message: reportName + " : ยังไม่ได้ระบุ Spreadsheet ที่ " + settingSheetName + "!" + urlCell,
        levels: {}
      };
    }

    const sourceId = extractSpreadsheetId_(url);
    const ss = sourceId
      ? SpreadsheetApp.openById(sourceId)
      : SpreadsheetApp.openByUrl(url);
    const levels = ["ประถม", "ม.ต้น", "ม.ปลาย"];
    const result = {};

    levels.forEach(function(level) {
      const sh = ss.getSheetByName(level);

      if (!sh) {
        result[level] = [];
        return;
      }

      const lastRow = sh.getLastRow();

      if (lastRow < 2) {
        result[level] = [];
        return;
      }

      const data = sh
        .getRange(2, teacherCol, lastRow - 1, 2)
        .getDisplayValues();

      const map = {};

      data.forEach(function(r) {
        const teacher = String(r[0] || "").trim();
        const district = String(r[1] || "").trim();

        if (!teacher) return;

        const key = teacher + "||" + district;

        if (!map[key]) {
          map[key] = {
            teacher: teacher,
            district: district,
            count: 0
          };
        }

        map[key].count++;
      });

      const total = Object.values(map).reduce(function(sum, x) {
        return sum + x.count;
      }, 0);

      result[level] = Object.values(map)
        .map(function(x) {
          return {
            teacher: x.teacher,
            district: x.district,
            count: x.count,
            percent: total ? ((x.count / total) * 100).toFixed(2) : "0.00"
          };
        })
        .sort(function(a, b) {
          return Number(b.percent) - Number(a.percent);
        });
    });

    return {
      ok: true,
      message: "",
      levels: result
    };

  } catch (err) {
    return {
      ok: false,
      message: reportName + " : " + err.message,
      levels: {}
    };
  }
}

    const lastRow = sourceSh.getLastRow();
    const lastCol = sourceSh.getLastColumn();

    if (lastRow < 1 || lastCol < 1) return;

    const values = sourceSh.getRange(1, 1, lastRow, lastCol).getDisplayValues();
    const targetSh = tempSS.insertSheet(prefix + "_" + level);

    targetSh.getRange(1, 1, values.length, values[0].length).setValues(values);

    targetSh.getRange(1, 1, 1, values[0].length)
      .setFontWeight("bold")
      .setBackground("#2e7d1f")
      .setFontColor("#ffffff");
  });
}

/**

 *   พยายามหารหัสนักศึกษาจากคอลัมน์ A/B เพื่อจับคู่รูป
 *
 * ทั้งสองประเภทคำนวณเปอร์เซ็นต์ = จำนวนรายการของชื่อนั้น /
 * จำนวนรายการทั้งหมดที่ผ่านเงื่อนไข Key ในระดับชั้นเดียวกัน × 100
 */
function getStudentServiceTop3_() {
  const cache = CacheService.getScriptCache();
  const cacheKey = 'STUDENT_SERVICE_TOP3_V7_TAMBOL_TOP3_HEADING_ORG';
  const cached = cache.get(cacheKey);
  if (cached) {
    try { return JSON.parse(cached); } catch (ignore) {}
  }

  const levels = ["ประถม", "ม.ต้น", "ม.ปลาย"];
  const emptyLevels_ = function() {
    return {
      "ประถม": [],
      "ม.ต้น": [],
      "ม.ปลาย": []
    };
  };

  const result = {
    success: true,
    generatedAt: new Date().toISOString(),
    organizationName: "",
    worksheet: emptyLevels_(),
    quiz: emptyLevels_()
  };

  let studentDirectory;
  try {
    studentDirectory = getStudentServiceStudentDirectory_();
  } catch (err) {
    // หากฐานรูปมีปัญหา อันดับยังต้องแสดงได้ เพียงใช้รูป fallback
    studentDirectory = {
      byId: {},
      byName: {},
      byLooseName: {}
    };
  }

  let reports;
  try {
    const mainSS = SpreadsheetApp.openById(DATABSE_MAIN_TEATHER_ID);

    // ชื่อหน่วยงานสำหรับกรอบ TOP 3 ของเว็บไซต์ตำบล
    // DATABSE_MAIN_TEATHER_ID > set_login!C3
    const setLoginSh = mainSS.getSheetByName('set_login');
    if (setLoginSh) {
      result.organizationName = String(setLoginSh.getRange('C3').getDisplayValue() || '').trim();
    }

    reports = {
      // ใบงาน: classroom!B9 -> กรอง Key ที่คอลัมน์ C -> นับชื่อคอลัมน์ D
      worksheet: getWorksheetStudentRankingBySiteKey_(mainSS, studentDirectory),
      // แบบทดสอบ: quiz!B8 -> กรอง Key ที่คอลัมน์ D -> นับชื่อคอลัมน์ B
      quiz: getQuizStudentRankingBySiteKey_(mainSS, studentDirectory)
    };
  } catch (err) {
    result.worksheet.error = "ใบงาน : " + (err && err.message ? err.message : err);
    result.quiz.error = "แบบทดสอบ : " + (err && err.message ? err.message : err);
    try { cache.put(cacheKey, JSON.stringify(result), 60); } catch (ignore) {}
    return result;
  }

  ["worksheet", "quiz"].forEach(function(type) {
    const report = reports[type] || {};

    if (!report.ok) {
      result[type].error = report.message || "โหลดข้อมูลไม่สำเร็จ";
      return;
    }

    levels.forEach(function(level) {
      result[type][level] = (report.levels[level] || [])
        .slice(0, 3)
        .map(function(row, index) {
          const enriched = enrichStudentServiceRankingRow_(row, studentDirectory);
          return {
            rank: index + 1,
            // teacher คงไว้เพื่อไม่กระทบ frontend/โค้ดเดิมที่อาจอ้างถึง
            teacher: String(row.teacher || ""),

            displayName: String(enriched.displayName || ""),
            studentId: String(enriched.studentId || row.studentId || ""),
            photoUrl: String(enriched.photoUrl || ""),
            district: String(row.district || ""),
            count: Number(row.count || 0),
            percent: Number(row.percent || 0)
          };
        });
    });
  });

  const hasError = Boolean(result.worksheet.error || result.quiz.error);
  try {
    cache.put(cacheKey, JSON.stringify(result), hasError ? 60 : 300);
  } catch (ignore) {}

  return result;
}

function normalizeStudentServiceName_(value) {
  return String(value || '')
    .replace(/[\u200B-\u200D\uFEFF]/g, '')
    .replace(/\s+/g, ' ')
    .trim()
    .toLowerCase();
}

function normalizeStudentServiceLooseName_(value) {
  return normalizeStudentServiceName_(value)
    .replace(/^(?:เด็กชาย|เด็กหญิง|ด\.ช\.|ด\.ญ\.|นางสาว|น\.ส\.|นาง|นาย|ครู)\s*/i, '')
    .replace(/\s+/g, ' ')
    .trim();
}

function getStudentServiceDisplayName_(value) {
  const name = String(value || '').trim().replace(/\s+/g, ' ');
  if (!name) return '';

  const parts = name.split(' ').filter(Boolean);
  if (parts.length <= 1) return name;

  // รองรับนามสกุลลักษณะ "ณ เชียงใหม่" โดยตัดทั้งส่วนสกุลออก
  if (parts.length >= 3 && parts[parts.length - 2] === 'ณ') {
    return parts.slice(0, -2).join(' ') || parts[0];
  }

  return parts.slice(0, -1).join(' ');
}

function getStudentServiceStudentDirectory_() {
  const directory = {
    byId: {},
    byName: {},
    byLooseName: {}
  };

  const spreadsheetId = String(DATABSE_STUDENT_LIST_ID || '').trim();
  if (!spreadsheetId) return directory;

  const ss = SpreadsheetApp.openById(spreadsheetId);
  const sh = ss.getSheetByName('รายชื่อนักศึกษา');
  if (!sh || sh.getLastRow() < 2) return directory;

  // B:F = รหัสนักศึกษา / รายชื่อนักศึกษา / ชั้น / PIN / รูปนักศึกษา
  const rows = sh.getRange(2, 2, sh.getLastRow() - 1, 5).getDisplayValues();
  const looseBuckets = {};

  rows.forEach(function(row) {
    const studentId = String(row[0] || '').trim(); // B
    const fullName = String(row[1] || '').trim();  // C
    const photoUrl = normalizePublicImageUrl_(String(row[4] || '').trim()); // F

    if (!studentId && !fullName) return;

    const item = {
      studentId: studentId,
      fullName: fullName,
      displayName: getStudentServiceDisplayName_(fullName),
      photoUrl: photoUrl
    };

    if (studentId) {
      directory.byId[studentId] = item;
    }

    const exactName = normalizeStudentServiceName_(fullName);
    if (exactName) {
      directory.byName[exactName] = item;
    }

    const looseName = normalizeStudentServiceLooseName_(fullName);
    if (looseName) {
      if (!looseBuckets[looseName]) looseBuckets[looseName] = [];
      looseBuckets[looseName].push(item);
    }
  });

  // loose name ใช้เฉพาะกรณีพบเพียงคนเดียว ป้องกันจับรูปผิดเมื่อชื่อซ้ำ
  Object.keys(looseBuckets).forEach(function(key) {
    if (looseBuckets[key].length === 1) {
      directory.byLooseName[key] = looseBuckets[key][0];
    }
  });

  return directory;
}

function findStudentServiceProfile_(studentId, fullName, directory) {
  directory = directory || { byId: {}, byName: {}, byLooseName: {} };

  const id = String(studentId || '').trim();
  if (id && directory.byId[id]) return directory.byId[id];

  const exactName = normalizeStudentServiceName_(fullName);
  if (exactName && directory.byName[exactName]) return directory.byName[exactName];

  const looseName = normalizeStudentServiceLooseName_(fullName);
  if (looseName && directory.byLooseName[looseName]) return directory.byLooseName[looseName];

  return null;
}

function findStudentServiceIdFromCells_(cells, directory) {
  const values = (cells || []).map(function(v) {
    return String(v || '').trim();
  }).filter(Boolean);

  // ให้รหัสที่ตรงกับฐานรายชื่อนักศึกษามีสิทธิ์ก่อน
  for (let i = 0; i < values.length; i++) {
    if (directory && directory.byId && directory.byId[values[i]]) {
      return values[i];
    }
  }

  // fallback: รูปแบบรหัสนักศึกษาที่เป็นตัวเลขยาว 8-15 หลัก
  for (let i = 0; i < values.length; i++) {
    if (/^\d{8,15}$/.test(values[i])) return values[i];
  }

  return '';
}

function enrichStudentServiceRankingRow_(row, directory) {
  const originalName = String(row.teacher || row.fullName || '').trim();
  const originalId = String(row.studentId || '').trim();
  const profile = findStudentServiceProfile_(originalId, originalName, directory);
  const fullName = profile && profile.fullName ? profile.fullName : originalName;

  return {
    studentId: profile && profile.studentId ? profile.studentId : originalId,
    fullName: fullName,
    displayName: profile && profile.displayName
      ? profile.displayName
      : getStudentServiceDisplayName_(fullName),
    photoUrl: profile && profile.photoUrl ? profile.photoUrl : ''
  };
}

/**
 * คำนวณอันดับการส่งใบงานเฉพาะข้อมูลของเว็บไซต์ตำบลนี้
 * 1) Key = ID_SHEET > Sheet1!C9
 * 2) URL ฐานคำตอบห้องเรียน/ใบงาน = DATABSE_MAIN_TEATHER_ID > classroom!B9
 * 3) ในชีต ประถม/ม.ต้น/ม.ปลาย: C=Key, D=ชื่อที่นำมาจัดอันดับ
 * 4) กรอง C === Key
 * 5) นับชื่อจาก D (ชื่อซ้ำ = เพิ่ม count, ชื่อไม่ซ้ำ = count 1)
 * 6) percent = count / จำนวนแถวที่ C ตรง Key และมีชื่อใน D ทั้งหมด × 100
 * 7) เรียง count มากไปน้อย; ถ้าเท่ากันให้ชื่อที่ปรากฏก่อนในชีตอยู่ก่อน
 */
function getWorksheetStudentRankingBySiteKey_(mainSS, studentDirectory) {
  try {
    const idSS = SpreadsheetApp.openById(ID_SHEET);
    const keySheet = idSS.getSheetByName('Sheet1');
    if (!keySheet) {
      return {
        ok: false,
        message: 'ใบงาน : ไม่พบชีต Sheet1 ใน ID_SHEET',
        levels: {}
      };
    }

    const siteKey = String(keySheet.getRange('C9').getDisplayValue() || '').trim();
    if (!siteKey) {
      return {
        ok: false,
        message: 'ใบงาน : ยังไม่ได้ระบุ Key ที่ ID_SHEET > Sheet1!C9',
        levels: {}
      };
    }

    mainSS = mainSS || SpreadsheetApp.openById(DATABSE_MAIN_TEATHER_ID);
    const settingSh = mainSS.getSheetByName('classroom');
    if (!settingSh) {
      return {
        ok: false,
        message: 'ใบงาน : ไม่พบชีตตั้งค่า classroom',
        levels: {}
      };
    }

    const url = String(settingSh.getRange('B9').getDisplayValue() || '').trim();
    if (!url) {
      return {
        ok: false,
        message: 'ใบงาน : ยังไม่ได้ระบุ Spreadsheet ที่ classroom!B9',
        levels: {}
      };
    }

    const sourceId = extractSpreadsheetId_(url);
    const answerSS = sourceId
      ? SpreadsheetApp.openById(sourceId)
      : SpreadsheetApp.openByUrl(url);

    const levels = ["ประถม", "ม.ต้น", "ม.ปลาย"];
    const result = {};

    levels.forEach(function(level) {
      const sh = answerSS.getSheetByName(level);
      if (!sh || sh.getLastRow() < 2) {
        result[level] = [];
        return;
      }

      // อ่าน A:D เพื่อใช้ C=Key, D=ชื่อ และทดลองจับรหัสนักศึกษาจาก A/B
      const rows = sh.getRange(2, 1, sh.getLastRow() - 1, 4).getDisplayValues();
      const map = {};
      let totalMatched = 0;

      rows.forEach(function(r, index) {
        const rowKey = String(r[2] || '').trim();       // C
        const studentName = String(r[3] || '').trim(); // D

        if (rowKey !== siteKey || !studentName) return;

        totalMatched++;

        if (!map[studentName]) {
          map[studentName] = {
            teacher: studentName,
            studentId: findStudentServiceIdFromCells_([r[0], r[1]], studentDirectory),
            district: siteKey,
            count: 0,
            firstRow: index + 2
          };
        } else if (!map[studentName].studentId) {
          map[studentName].studentId = findStudentServiceIdFromCells_([r[0], r[1]], studentDirectory);
        }

        map[studentName].count++;
      });

      result[level] = Object.values(map)
        .map(function(item) {
          return {
            teacher: item.teacher,
            studentId: item.studentId,
            district: item.district,
            count: item.count,
            firstRow: item.firstRow,
            percent: totalMatched
              ? ((item.count / totalMatched) * 100).toFixed(2)
              : '0.00'
          };
        })
        .sort(function(a, b) {
          const countDiff = Number(b.count) - Number(a.count);
          if (countDiff !== 0) return countDiff;
          return Number(a.firstRow) - Number(b.firstRow);
        })
        .map(function(item) {
          return {
            teacher: item.teacher,
            studentId: item.studentId,
            district: item.district,
            count: item.count,
            percent: item.percent
          };
        });
    });

    return {
      ok: true,
      message: '',
      key: siteKey,
      levels: result
    };

  } catch (err) {
    return {
      ok: false,
      message: 'ใบงาน : ' + err.message,
      levels: {}
    };
  }
}

function getTargetSpreadsheetId_() {

function getPublicData() {
  const spreadsheetId = String(DATABSE_MAIN_TEATHER_ID || '').trim();
  if (!spreadsheetId) throw new Error('ไม่ได้กำหนด DATABSE_MAIN_TEATHER_ID');

  // PERFORMANCE: เปิด Spreadsheet เพียงครั้งเดียว และรวมสถิติ visits/reviews แบบ bulk
  const ss = SpreadsheetApp.openById(spreadsheetId);

  function readRows_(sheet) {
    if (!sheet || sheet.getLastRow() < 2 || sheet.getLastColumn() < 1) return [];
    const values = sheet.getDataRange().getDisplayValues();
    const headers = values[0].map(function(value) { return String(value || '').trim(); });
    return values.slice(1)
      .filter(function(row) {
        return row.some(function(value) { return String(value || '').trim() !== ''; });
      })
      .map(function(row) {
        const item = {};
        headers.forEach(function(header, index) { item[header] = row[index]; });
        return item;
      });
  }

  const settingsSheet = ss.getSheetByName(CONFIG.SHEETS.SETTINGS);
  const settings = {};
  readRows_(settingsSheet).forEach(function(row) {
    const key = String(row.key || '').trim();
    if (key) settings[key] = row.value;
  });

  const typeColorMap = {};
  const types = [];
  const typeSheet = ss.getSheetByName('type');
  if (typeSheet && typeSheet.getLastRow() >= 1) {
    typeSheet.getRange(1, 1, typeSheet.getLastRow(), 2).getDisplayValues().forEach(function(row) {
      const category = String(row[0] || '').trim();
      const color = String(row[1] || '').trim();
      if (!category) return;
      types.push(category);
      typeColorMap[category] = color || '#FF3300';
    });
  }

  const sourceSheet = ss.getSheetByName(CONFIG.SHEETS.SOURCES);
  if (!sourceSheet) throw new Error('ไม่พบชีต ' + CONFIG.SHEETS.SOURCES);
  const sourceRows = readRows_(sourceSheet).filter(function(row) {
    return String(row.status || '').trim().toLowerCase() === 'block';
  });
  const visitRows = readRows_(ss.getSheetByName(CONFIG.SHEETS.VISITS));
  const reviewRows = readRows_(ss.getSheetByName(CONFIG.SHEETS.REVIEWS));
  const visitCountBySource = {};
  const reviewsBySource = {};

  visitRows.forEach(function(row) {
    const id = String(row.source_id || '').trim();
    if (id) visitCountBySource[id] = (visitCountBySource[id] || 0) + 1;
  });
  reviewRows.forEach(function(row) {
    if (String(row.status || '').trim() !== 'approved') return;
    const id = String(row.source_id || '').trim();
    if (!id) return;
    if (!reviewsBySource[id]) reviewsBySource[id] = [];
    reviewsBySource[id].push(row);
  });

  const sources = sourceRows.map(function(source) {
    const id = String(source.id || '').trim();
    const reviews = reviewsBySource[id] || [];
    const category = String(source.category || '').trim();
    return Object.assign({}, source, {
      views: visitCountBySource[id] || 0,
      averageRating: average_(reviews.map(function(row) { return Number(row.rating); }).filter(Boolean)),
      reviewCount: reviews.length,
      reviews: reviews.slice(-10).reverse(),
      marker_color: typeColorMap[category] || '#FF3300'
    });
  });
  const areas = [...new Set(sources.map(function(source) {
    return String(source.area || '').trim();
  }).filter(Boolean))].sort(function(a, b) { return a.localeCompare(b, 'th'); });

  return { success: true, settings: settings, sources: sources, types: types, areas: areas };
}
function getSourceDetail(
  sourceId,
  visitType,
  sessionId,
  userAgent
) {
  // รายละเอียดหน้าบ้านใช้ DATABSE_MAIN_TEATHER_ID เช่นเดียวกับหน้า /exec
  const spreadsheetId = String(DATABSE_MAIN_TEATHER_ID || '').trim();

  if (!spreadsheetId) {
    throw new Error('ไม่ได้กำหนด DATABSE_MAIN_TEATHER_ID');
  }

  const source = getSheetObjects_(
    CONFIG.SHEETS.SOURCES,
    spreadsheetId
  )
    .find(r => String(r.id) === String(sourceId));

  if (!source) {
    throw new Error('ไม่พบข้อมูลแหล่งเรียนรู้');
  }

  if (visitType) {
    recordVisit_(

      userAgent || '',
      spreadsheetId
    );
  }

  return {
    success: true,
    source: enrichSource_(source, spreadsheetId)
  };
}
function submitReview(payload) {
  payload = payload || {};
  const sourceId = cleanText_(payload.sourceId, 50);
  const rating = Number(payload.rating);
  // ความคิดเห็นจากหน้าบ้านบันทึกลง DATABSE_MAIN_TEATHER_ID เพียงจุดเดียว
  const spreadsheetId = String(DATABSE_MAIN_TEATHER_ID || '').trim();

  if (!sourceId) throw new Error('ไม่พบรหัสแหล่งเรียนรู้');
  if (!rating || rating < 1 || rating > 5) throw new Error('กรุณาให้คะแนน 1-5 ดาว');
  if (!spreadsheetId) throw new Error('ไม่พบ Spreadsheet ID ของพื้นที่');

  const source = getSheetObjects_(
    CONFIG.SHEETS.SOURCES,
    spreadsheetId
  )
    .find(r => String(r.id) === sourceId);
  if (!source) throw new Error('ไม่พบข้อมูลแหล่งเรียนรู้');

  const lock = LockService.getScriptLock();

  if (!lock.tryLock(10000)) {
    throw new Error('ระบบกำลังบันทึกข้อมูล กรุณาลองอีกครั้ง');
  }

  try {
    const sheet = getSheet_(CONFIG.SHEETS.REVIEWS, spreadsheetId);

    sheet.appendRow([
      'RV' + Utilities.getUuid().replace(/-/g, '').slice(0, 12).toUpperCase(),
      new Date(),
      source.id,
      source.name,
      cleanText_(payload.name, 100),
      rating,
      cleanText_(payload.comment, 1000),
      'approved'
    ]);

    SpreadsheetApp.flush();
  } finally {
    lock.releaseLock();
  }

  return { success: true, message: 'บันทึกความคิดเห็นเรียบร้อยแล้ว' };
}

function getAdminDashboard(pin) {

  const adminSpreadsheetId = DATABSE_MAIN_TEATHER_ID;
  const sources = getSheetObjects_(CONFIG.SHEETS.SOURCES, adminSpreadsheetId)
    .map(source => enrichSource_(source, adminSpreadsheetId));
  const visits = getSheetObjects_(CONFIG.SHEETS.VISITS, adminSpreadsheetId);
  const reviews = getSheetObjects_(CONFIG.SHEETS.REVIEWS, adminSpreadsheetId);

  const todayKey = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd');
  const monthKey = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM');

  const categoryCounts = {};
  const sourceVisitCounts = {};
  const dailyCounts = {};

  sources.forEach(s => categoryCounts[s.category] = (categoryCounts[s.category] || 0) + 1);
  visits.forEach(v => {
    sourceVisitCounts[v.source_id] = (sourceVisitCounts[v.source_id] || 0) + 1;
    dailyCounts[v.date_key] = (dailyCounts[v.date_key] || 0) + 1;
  });

  const popular = sources
    .map(s => ({ id: s.id, name: s.name, views: sourceVisitCounts[s.id] || 0 }))
    .sort((a, b) => b.views - a.views)
    .slice(0, 5);

  return {
    success: true,
    settings: getSettings_(adminSpreadsheetId),
    summary: {
      totalSources: sources.length,
      totalVisits: visits.length,
      todayVisits: visits.filter(v => String(v.date_key) === todayKey).length,
      monthVisits: visits.filter(v => String(v.month_key) === monthKey).length,
      totalReviews: reviews.length,
      averageRating: average_(reviews.map(r => Number(r.rating)).filter(Boolean))
    },
    sources,
    reviews: reviews.slice(-50).reverse(),
    categoryCounts,
    popular,
    dailyCounts
  };
}

function getAdminLearningImageFolderId_() {
  const selectorSS =
    SpreadsheetApp.openById(
      ID_SHEET
    );

  const selectorSheet =
    selectorSS.getSheetByName(
      ADMIN_USER_SELECTOR_SHEET_NAME
    );

  if (!selectorSheet) {
    throw new Error(
      'ไม่พบชีต ' +
      ADMIN_USER_SELECTOR_SHEET_NAME +
      ' สำหรับอ่าน Folder ID'
    );
  }

  const rawValue = String(
    selectorSheet
      .getRange('C8')
      .getDisplayValue() || ''
  ).trim();

  if (!rawValue) {
    throw new Error(
      'ยังไม่ได้ระบุ Folder ID ที่ ' +
      ADMIN_USER_SELECTOR_SHEET_NAME +
      '!C8'
    );
  }

  // รองรับทั้ง Folder ID ตรง ๆ และ URL ของ Google Drive
  const idMatch =
    rawValue.match(/[-\w]{20,}/);

  if (!idMatch) {
    throw new Error(
      'Folder ID ที่ ' +
      ADMIN_USER_SELECTOR_SHEET_NAME +
      '!C8 ไม่ถูกต้อง'
    );
  }

  return idMatch[0];
}

  );

  if (!match) {
    throw new Error('ข้อมูลรูปภาพไม่ถูกต้อง');
  }

  const mimeType =
    String(match[1] || '').toLowerCase();

  const allowedMimeTypes = [
    'image/jpeg',
    'image/png',
    'image/webp',
    'image/gif'
  ];

  if (allowedMimeTypes.indexOf(mimeType) === -1) {
    throw new Error(
      'รองรับเฉพาะไฟล์ JPG, PNG, WEBP และ GIF'
    );
  }

  let bytes;

  try {
    bytes = Utilities.base64Decode(match[2]);
  } catch (err) {
    throw new Error('ไม่สามารถถอดรหัสไฟล์รูปภาพได้');
  }

  const maxBytes = 10 * 1024 * 1024;

  if (bytes.length > maxBytes) {
    throw new Error(
      'ไฟล์รูปภาพต้องมีขนาดไม่เกิน 10 MB'
    );
  }

  const folderId =
    getAdminLearningImageFolderId_();

  const original =
    String(originalName || 'image.jpg')
      .trim()
      .replace(/[\\/:*?"<>|]+/g, '_')
      .replace(/\s+/g, '-')
      .slice(-120) ||
    'image.jpg';

  const safePrefix =
    String(prefix || 'learning-image')
      .trim()
      .replace(/[^a-zA-Z0-9_-]+/g, '-')
      .replace(/^-+|-+$/g, '')
      .slice(0, 50) ||
    'learning-image';

  const safeName =
    safePrefix +
    '-' +
    Date.now() +
    '-' +
    original;

  const blob =
    Utilities.newBlob(
      bytes,
      mimeType,
      safeName
    );

  let file;

  try {
    // Advanced Drive Service - Drive API V3
    file = Drive.Files.create(
      {
        name: safeName,
        parents: [folderId]
      },
      blob,
      {
        supportsAllDrives: true,
        fields: 'id,name'
      }
    );
  } catch (err) {
    throw new Error(
      'อัปโหลดรูปเข้า Google Drive ไม่สำเร็จ: ' +
      (err && err.message ? err.message : err)
    );
  }

  if (!file || !file.id) {
    throw new Error(
      'Google Drive ไม่ส่ง File ID กลับมา'
    );
  }

  try {
    // Advanced Drive Service - Drive API V3
    Drive.Permissions.create(
      {
        type: 'anyone',
        role: 'reader'
      },
      file.id,
      {
        supportsAllDrives: true
      }
    );
  } catch (err) {
    throw new Error(
      'อัปโหลดไฟล์แล้ว แต่ตั้งค่าสิทธิ์ดูรูปไม่สำเร็จ: ' +
      (err && err.message ? err.message : err)
    );
  }

  return {
    success: true,
    fileId: file.id,
    url:
      'https://lh3.googleusercontent.com/d/' +
      file.id,
    message: 'อัปโหลดรูปภาพเรียบร้อยแล้ว'
  };
}

function saveSource(pin, payload) {
  payload = payload || {};

  const sheet = getSheet_(CONFIG.SHEETS.SOURCES, DATABSE_MAIN_TEATHER_ID);
  const data = sheet.getDataRange().getValues();
  const headers = data[0].map(String);
  const idIndex = headers.indexOf('id');

  let id = cleanText_(payload.id, 50);
  if (!id) id = createNextSourceId_(data.slice(1).map(r => String(r[idIndex] || '')));

const values = {
  id,

  name: cleanText_(payload.name, 200),

  image_url:
    cleanText_(payload.image_url, 1000),

  image2_url:
    cleanText_(payload.image2_url, 1000),

  image3_url:
    cleanText_(payload.image3_url, 1000),

  manager: cleanText_(payload.manager, 200),
  phone: cleanText_(payload.phone, 50),
  address: cleanText_(payload.address, 500),
  area: cleanText_(payload.area, 200),
  category: cleanText_(payload.category, 200),

  description:
    cleanText_(payload.description, 3000),

  learn_more_url:
    cleanText_(payload.learn_more_url, 2000),

  latitude: cleanText_(payload.latitude, 50),
  longitude: cleanText_(payload.longitude, 50),

  map_x: clamp_(Number(payload.map_x), 0, 100),
  map_y: clamp_(Number(payload.map_y), 0, 100),

  marker_icon:
    cleanText_(payload.marker_icon, 100) ||
    'fa-location-dot',

  status:
    String(payload.status || 'block') === 'none'
      ? 'none'
      : 'block',

  updated_at: new Date()
};

  if (!values.name) throw new Error('กรุณาระบุชื่อแหล่งเรียนรู้');

  const rowIndex = data.findIndex((r, index) => index > 0 && String(r[idIndex]) === id);

  if (rowIndex > 0) {
    headers.forEach((h, col) => {
      if (Object.prototype.hasOwnProperty.call(values, h)) {
        sheet.getRange(rowIndex + 1, col + 1).setValue(values[h]);
      }
    });
  } else {
    values.created_at = new Date();
    sheet.appendRow(headers.map(h => Object.prototype.hasOwnProperty.call(values, h) ? values[h] : ''));
  }

  return { success: true, id, message: 'บันทึกข้อมูลเรียบร้อยแล้ว' };
}

function recordVisit_(
  source,
  visitType,
  sessionId,
  userAgent,
  spreadsheetId
) {
  const lock = LockService.getScriptLock();
  lock.tryLock(5000);

  try {
    const cache = CacheService.getScriptCache();

    const dedupeKey = [
      'visit',
      spreadsheetId,
      source.id,
      visitType,
      sessionId
    ].join(':');

    if (sessionId && cache.get(dedupeKey)) return;

    const now = new Date();

    getSheet_(
      CONFIG.SHEETS.VISITS,
      spreadsheetId
    ).appendRow([
      'VS' +
        Utilities.getUuid()
          .replace(/-/g, '')
          .slice(0, 12)
          .toUpperCase(),
      now,
      source.id,
      source.name,
      cleanText_(visitType, 30),
      cleanText_(sessionId, 100),
      detectDevice_(userAgent),
      cleanText_(userAgent, 500),
      Utilities.formatDate(
        now,
        Session.getScriptTimeZone(),
        'yyyy-MM-dd'
      ),
      Utilities.formatDate(
        now,
        Session.getScriptTimeZone(),
        'yyyy-MM'
      )
    ]);

    if (sessionId) {
      cache.put(dedupeKey, '1', 1800);
    }

  } finally {
    try {
      lock.releaseLock();
    } catch (e) {}
  }
}

function enrichSource_(
  source,
  spreadsheetId
) {

  const visits = getSheetObjects_(
    CONFIG.SHEETS.VISITS,
    spreadsheetId
  )
    .filter(v =>
      String(v.source_id) ===
      String(source.id)
    );

  const reviews = getSheetObjects_(
    CONFIG.SHEETS.REVIEWS,
    spreadsheetId
  )
    .filter(r =>
      String(r.source_id) ===
        String(source.id) &&
      String(r.status) === 'approved'
    );

  return Object.assign({}, source, {
    views: visits.length,

    averageRating: average_(
      reviews
        .map(r => Number(r.rating))
        .filter(Boolean)
    ),

    reviewCount: reviews.length,
    reviews: reviews.slice(-10).reverse()
  });
}

function getSheetObjects_(name, spreadsheetId) {
  const values = getSheet_(name, spreadsheetId)
    .getDataRange()
    .getDisplayValues();

  if (values.length < 2) return [];

  const headers = values[0].map(String);

  return values
    .slice(1)
    .filter(row =>
      row.some(v => String(v).trim() !== '')
    )
    .map(row =>
      headers.reduce((obj, h, i) => {
        obj[h] = row[i];
        return obj;
      }, {})
    );
}

  if (sheet.getLastRow() === 0 && rows && rows.length) {
    sheet.getRange(1, 1, rows.length, rows[0].length).setValues(rows);

  return sheet;
}

function formatSheets_(ss) {
  Object.values(CONFIG.SHEETS).forEach(name => {
    const sheet = ss.getSheetByName(name);
    if (!sheet || sheet.getLastRow() === 0) return;
    sheet.setFrozenRows(1);
    sheet.getRange(1, 1, 1, sheet.getLastColumn())
      .setFontWeight('bold')
      .setBackground('#1e3a8a')
      .setFontColor('#ffffff');
    sheet.autoResizeColumns(1, sheet.getLastColumn());
  });
}

function createNextSourceId_(ids) {
  const max = ids.reduce((m, id) => {
    const n = Number(String(id).replace(/\D/g, '')) || 0;
    return Math.max(m, n);
  }, 0);
  return 'LR' + String(max + 1).padStart(4, '0');
}

function cleanText_(value, maxLength) {
  return String(value == null ? '' : value).trim().slice(0, maxLength || 1000);
}

function clamp_(value, min, max) {
  if (!Number.isFinite(value)) return 50;
  return Math.min(max, Math.max(min, value));
}

function average_(numbers) {
  if (!numbers.length) return 0;
  return Math.round((numbers.reduce((a, b) => a + b, 0) / numbers.length) * 10) / 10;
}

function detectDevice_(ua) {
  ua = String(ua || '').toLowerCase();
  if (/ipad|tablet/.test(ua)) return 'Tablet';
  if (/mobile|android|iphone/.test(ua)) return 'Mobile';
  return 'Desktop';
}

function getLearningAreas_() {
  const spreadsheetId = DATABSE_MAIN_TEATHER_ID;
  const settings = getSettings_(spreadsheetId) || {};
  const typeColorMap = getTypeColorMap_(spreadsheetId) || {};

  // อ่านทุกแถวจาก learning_sources โดยรองรับหัวตารางทั้งแบบเดิม/ใหม่
  // และไม่บังคับว่า status ต้องเป็นคำว่า block เท่านั้น
  // ซ่อนเฉพาะรายการที่กำหนด status = none อย่างชัดเจน
  const sourceSheet = getSheet_(CONFIG.SHEETS.SOURCES, spreadsheetId);
  const sourceRows = sheetRowsAsObjects_(sourceSheet)
    .map(function(row, index) {
      return normalizeLearningSource_(row, index);
    })
    .filter(function(row) {
      const status = String(row.status || 'block').trim().toLowerCase();
      if (status === 'none') return false;
      return Boolean(
        String(row.id || '').trim() ||
        String(row.name || '').trim() ||
        String(row.image_url || '').trim() ||
        String(row.description || '').trim()
      );
    });

  const visitRows = getSheetObjects_(CONFIG.SHEETS.VISITS, spreadsheetId);
  const reviewRows = getSheetObjects_(CONFIG.SHEETS.REVIEWS, spreadsheetId)
    .filter(function(row) {
      return String(row.status || 'approved').trim().toLowerCase() === 'approved';
    });

  const visitCountBySource = {};
  visitRows.forEach(function(row) {
    const sourceId = String(row.source_id || '').trim();
    if (!sourceId) return;
    visitCountBySource[sourceId] = (visitCountBySource[sourceId] || 0) + 1;
  });

  const ratingBySource = {};
  reviewRows.forEach(function(row) {
    const sourceId = String(row.source_id || '').trim();
    const rating = Number(String(row.rating || '').replace(/,/g, '').trim());
    if (!sourceId || !Number.isFinite(rating) || rating <= 0) return;
    if (!ratingBySource[sourceId]) ratingBySource[sourceId] = [];
    ratingBySource[sourceId].push(rating);
  });

  const sources = sourceRows.map(function(row, index) {
    const sourceId = String(row.id || ('LR' + String(index + 1).padStart(4, '0'))).trim();
    const ratings = ratingBySource[sourceId] || [];
    const averageRating = ratings.length
      ? Math.round((ratings.reduce(function(sum, value) { return sum + value; }, 0) / ratings.length) * 10) / 10
      : 0;

    const rawX = Number(String(row.map_x || '').replace(/,/g, '').trim());
    const rawY = Number(String(row.map_y || '').replace(/,/g, '').trim());
    const mapX = Number.isFinite(rawX) ? Math.max(0, Math.min(100, rawX)) : 50;
    const mapY = Number.isFinite(rawY) ? Math.max(0, Math.min(100, rawY)) : 50;
    const category = String(row.category || '').trim();
    const markerColor = String(row.marker_color || typeColorMap[category] || '#ef4444').trim();

    return {
      id: sourceId,
      name: String(row.name || '').trim(),
      category: category,
      manager: String(row.manager || '').trim(),
      phone: String(row.phone || '').trim(),
      address: String(row.address || '').trim(),
      area: String(row.area || '').trim(),
      description: String(row.description || '').trim(),
      image_url: normalizePublicImageUrl_(String(row.image_url || '').trim()),
      image2_url: normalizePublicImageUrl_(String(row.image2_url || '').trim()),
      image3_url: normalizePublicImageUrl_(String(row.image3_url || '').trim()),
      learn_more_url: String(row.learn_more_url || '').trim(),
      latitude: String(row.latitude || '').trim(),
      longitude: String(row.longitude || '').trim(),
      map_x: mapX,
      map_y: mapY,
      marker_color: isValidHexColor_(markerColor) ? markerColor : '#ef4444',
      marker_icon: String(row.marker_icon || 'fa-location-dot').trim(),
      views: visitCountBySource[sourceId] || 0,
      averageRating: averageRating,
      reviewCount: ratings.length
    };
  }).filter(function(source) {
    return source.name || source.image_url || source.description;
  });

  const allRatings = [];
  Object.keys(ratingBySource).forEach(function(sourceId) {
    ratingBySource[sourceId].forEach(function(value) { allRatings.push(value); });
  });

  const overallRating = allRatings.length
    ? Math.round((allRatings.reduce(function(sum, value) { return sum + value; }, 0) / allRatings.length) * 10) / 10
    : 0;

  return [{
    name: String(settings.organization || settings.system_name || 'แหล่งเรียนรู้').trim(),
    spreadsheetId: spreadsheetId,
    mapImage: normalizePublicImageUrl_(String(settings.map_image_url || '').trim()),
    mapTitle: String(settings.map_title || 'แผนที่แหล่งเรียนรู้').trim(),
    sourceCount: sources.length,
    rating: overallRating,
    sources: sources
  }];
}

  const ss = SpreadsheetApp.openById(DATABSE_MAIN_ID);
  let sheet = ss.getSheetByName(PERFORMANCE_REPORT_SHEET_NAME);
  const headers = ['ลำดับ','เรื่อง','ผู้รายงาน','ประเภทงาน','URL ไฟล์','คีย์ผู้โพสต์'];

  if (!sheet) sheet = ss.insertSheet(PERFORMANCE_REPORT_SHEET_NAME);
  if (sheet.getMaxColumns() < 6) {
    sheet.insertColumnsAfter(sheet.getMaxColumns(), 6 - sheet.getMaxColumns());
  }

  if (sheet.getLastRow() === 0) {
    sheet.getRange(1, 1, 1, 6).setValues([headers]);
  } else {
    const current = sheet.getRange(1, 1, 1, 6).getDisplayValues()[0];
    if (current.every(function(v){ return !String(v || '').trim(); })) {
      sheet.getRange(1, 1, 1, 6).setValues([headers]);
    } else {
      if (!String(current[0] || '').trim()) sheet.getRange(1, 1).setValue(headers[0]);
      if (!String(current[1] || '').trim()) sheet.getRange(1, 2).setValue(headers[1]);
      if (!String(current[2] || '').trim()) sheet.getRange(1, 3).setValue(headers[2]);
      if (!String(current[3] || '').trim()) sheet.getRange(1, 4).setValue(headers[3]);
      if (!String(current[4] || '').trim()) sheet.getRange(1, 5).setValue(headers[4]);
      if (String(current[5] || '').trim() !== headers[5]) sheet.getRange(1, 6).setValue(headers[5]);
    }
  }

  // แถวเดิมในฐานกลางเป็นข้อมูลเดิมของเว็บไซต์อำเภอ
  const lastRow = sheet.getLastRow();
  const districtKey = getDistrictContentPosterKey_();
  if (lastRow >= 2 && districtKey) {
    const baseRows = sheet.getRange(2, 1, lastRow - 1, 5).getDisplayValues();
    const keyRows = sheet.getRange(2, 6, lastRow - 1, 1).getDisplayValues();
    let changed = false;
    const next = keyRows.map(function(row, index) {
      const currentKey = String((row && row[0]) || '').trim();
      if (currentKey) return [currentKey];
      const hasData = (baseRows[index] || []).some(function(v) {
        return String(v || '').trim() !== '';
      });
      if (hasData) {
        changed = true;
        return [districtKey];
      }
      return [''];
    });
    if (changed) {
      sheet.getRange(2, 6, next.length, 1).setValues(next);
      SpreadsheetApp.flush();
    }
  }

  return sheet;
}

function readPerformanceReportItems_(includeRowNumber) {
  const sheet = getPerformanceReportAdminSheet_();
  if (!sheet || sheet.getLastRow() < 2) return [];

  const rows = sheet.getRange(2, 1, sheet.getLastRow() - 1, 6).getDisplayValues();
  return rows.map(function(row, index) {
    const rawOrder = String(row[0] || '').trim();
    const numericOrder = Number(rawOrder);
    return {
      rowNumber: includeRowNumber ? index + 2 : 0,
      order: rawOrder,
      title: String(row[1] || '').trim(),
      reporter: String(row[2] || '').trim(),
      type: String(row[3] || '').trim(),
      fileUrl: String(row[4] || '').trim(),
      posterKey: String(row[5] || '').trim(),
      _sortOrder: isFinite(numericOrder) ? numericOrder : index + 1,
      _rowIndex: index + 2
    };
  }).filter(function(item) {
    return item.order || item.title || item.reporter || item.type || item.fileUrl;
  });
}

function sortPerformanceReportItems_(items) {
  return (items || []).sort(function(a, b) {
    if (b._sortOrder !== a._sortOrder) return b._sortOrder - a._sortOrder;
    return b._rowIndex - a._rowIndex;
  });
}

function stripPerformanceReportSortFields_(item) {
  return {
    rowNumber: Number(item.rowNumber || 0),
    order: String(item.order || '').trim(),
    title: String(item.title || '').trim(),
    reporter: String(item.reporter || '').trim(),
    type: String(item.type || '').trim(),
    fileUrl: String(item.fileUrl || '').trim(),
    posterKey: String(item.posterKey || '').trim()
  };
}

function getPerformanceReportPublicItems_() {
  const posterKey = getContentPosterKey_();
  const items = readPerformanceReportItems_(false).filter(function(item) {
    if (posterKey && String(item.posterKey || '').trim() !== posterKey) return false;
    return true;
  });

  return sortPerformanceReportItems_(items).map(stripPerformanceReportSortFields_);
}

function getPerformanceReportAdminData_() {
  const posterKey = getContentPosterKey_();
  const items = readPerformanceReportItems_(true)
    .filter(function(item) {
      return String(item.posterKey || '').trim() === posterKey;
    })
    .map(function(item) {
      // ใน popup ผู้รายงานต้องเป็นคีย์ของเว็บไซต์เสมอ
      item.reporter = posterKey;
      return item;
    });

  return {
    posterKey: posterKey,
    items: sortPerformanceReportItems_(items).map(stripPerformanceReportSortFields_)
  };
}

function getNextPerformanceReportOrder_(sheet) {
  if (sheet.getLastRow() < 2) return 1;
  const values = sheet.getRange(2, 1, sheet.getLastRow() - 1, 1).getDisplayValues().flat();
  let maxOrder = 0;
  values.forEach(function(value) {
    const n = Number(String(value || '').trim());
    if (isFinite(n) && n > maxOrder) maxOrder = n;
  });
  return maxOrder + 1;
}

function savePerformanceReportAdmin_(data) {
  data = data || {};
  const posterKey = getContentPosterKey_();
  const title = String(data.title || '').trim();
  const reporter = posterKey;
  const type = String(data.type || '').trim();
  let fileUrl = String(data.fileUrl || '').trim();

  if (!title) throw new Error('กรุณากรอกเรื่อง');
  if (!reporter) throw new Error('ไม่พบคีย์ผู้รายงานของเว็บไซต์');
  if (PERFORMANCE_REPORT_TYPES.indexOf(type) === -1) throw new Error('ประเภทงานไม่ถูกต้อง');

  if (String(data.fileData || '').trim()) {
    fileUrl = uploadContentAdminFile_(
      data.fileData,
      String(data.fileName || 'report-file'),
      'report-file',
      false
    );
  }

  if (!fileUrl) throw new Error('กรุณาระบุ URL หรืออัปโหลดไฟล์');
  if (!/^https?:\/\//i.test(fileUrl)) throw new Error('URL ไฟล์ไม่ถูกต้อง');

  const sheet = getPerformanceReportAdminSheet_();
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const requestedRow = Number(data.rowNumber || 0);
    const editing = requestedRow >= 2 && requestedRow <= sheet.getLastRow();
    let rowNumber = editing ? requestedRow : Math.max(2, sheet.getLastRow() + 1);
    let order = editing ? String(sheet.getRange(rowNumber, 1).getDisplayValue() || '').trim() : '';
    if (!order) order = String(getNextPerformanceReportOrder_(sheet));

    if (editing) {
      const existingKey = String(sheet.getRange(rowNumber, 6).getDisplayValue() || '').trim();
      if (existingKey !== posterKey) {
        throw new Error('ไม่อนุญาตให้แก้ไขรายงานของคีย์อื่น');
      }
      sheet.getRange(rowNumber, 1, 1, 6).setValues([[order, title, reporter, type, fileUrl, posterKey]]);
    } else {
      sheet.appendRow([order, title, reporter, type, fileUrl, posterKey]);
      rowNumber = sheet.getLastRow();
    }

    SpreadsheetApp.flush();
    return getPerformanceReportAdminData_();
  } finally {
    lock.releaseLock();
  }
}

function deletePerformanceReportAdmin_(rowNumber) {
  const posterKey = getContentPosterKey_();
  const sheet = getPerformanceReportAdminSheet_();
  const row = Number(rowNumber || 0);
  if (row < 2 || row > sheet.getLastRow()) throw new Error('ไม่พบรายงานที่ต้องการลบ');

  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    if (row > sheet.getLastRow()) throw new Error('ไม่พบรายงานที่ต้องการลบ');

    const existingKey = String(sheet.getRange(row, 6).getDisplayValue() || '').trim();
    if (existingKey !== posterKey) {
      throw new Error('ไม่อนุญาตให้ลบรายงานของคีย์อื่น');
    }

    sheet.deleteRow(row);
    SpreadsheetApp.flush();
    return getPerformanceReportAdminData_();
  } finally {
    lock.releaseLock();
  }
}

function runPerformanceReportAdminAction_(action, data) {
  const cmd = String(action || '').trim().toLowerCase();
  if (cmd === 'list') return getPerformanceReportAdminData_();
  if (cmd === 'save') return savePerformanceReportAdmin_(data || {});
  if (cmd === 'delete') return deletePerformanceReportAdmin_((data || {}).rowNumber);
  throw new Error('คำสั่งจัดการรายงานไม่ถูกต้อง');
}
